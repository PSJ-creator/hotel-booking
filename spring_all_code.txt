public interface AccomLikeRepository extends JpaRepository<AccomLike, Long> {
    boolean existsByMember_MemberIdAndAccommodation_ContentId(Long memberId, String contentId);

    Optional<AccomLike> findByMember_MemberIdAndAccommodation_ContentId(Long memberId, String contentId);

    List<AccomLike> findByMember_MemberIdOrderByCreatedAtDesc(Long memberId);

    // 💡 날짜가 있으면 재고가 있는 찜 숙소만 필터링, 날짜가 없으면(CAST IS NULL) 전체 찜 숙소 반환
    @Query("SELECT new org.example.hotelbooking.domain.accommodation.dto.StayResponseDto(" +
            "a.contentId, a.host.memberId, a.title, a.lodgingType, a.tel, a.zipcode, a.addr, " +
            "a.checkinTime, a.checkoutTime, a.roomCount, a.roomType, a.parking, a.subFacility, " +
            "a.overview, a.firstImage, a.firstImage2, a.mapX, a.mapY, " +
            "COALESCE((SELECT MIN(r.pricePerNight) FROM Room r WHERE r.accommodation = a), 0), " +
            "COALESCE((SELECT SUM(ri.availableCount) FROM RoomInventory ri JOIN ri.room r " +
            "          WHERE r.accommodation = a " +
            "            AND ri.inventoryDate = :checkIn " +
            "            AND (:guests IS NULL OR :guests <= 0 OR CAST(COALESCE(r.roomMaxCount, '0') AS integer) >= :guests)), 0L)) " +
            "FROM AccomLike al " +
            "JOIN al.accommodation a " +
            "WHERE al.member.memberId = :memberId " +
            // 💡 1. 인원 조건
            "  AND (:guests IS NULL OR :guests <= 0 OR EXISTS (" +
            "       SELECT 1 FROM Room r " +
            "       WHERE r.accommodation = a " +
            "         AND CAST(COALESCE(r.roomMaxCount, '0') AS integer) >= :guests)) " +
            // 💡 2. 날짜 조건 (:stayDays = 0L 이면 전체 찜 숙소 그대로 반환!)
            "  AND (:stayDays = 0L OR EXISTS (" +
            "       SELECT 1 FROM Room r " +
            "       WHERE r.accommodation = a " +
            "         AND (:guests IS NULL OR :guests <= 0 OR CAST(COALESCE(r.roomMaxCount, '0') AS integer) >= :guests) " +
            "         AND (SELECT COUNT(ri) FROM RoomInventory ri " +
            "              WHERE ri.room = r " +
            "                AND ri.inventoryDate >= :checkIn " +
            "                AND ri.inventoryDate < :checkOut " +
            "                AND ri.availableCount > 0) = :stayDays)) " +
            "ORDER BY al.createdAt DESC")

package org.example.hotelbooking.domain.like.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.dto.StayResponseDto;
import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.accommodation.repository.AccommodationRepository;
import org.example.hotelbooking.domain.like.entity.AccomLike;
import org.example.hotelbooking.domain.like.repository.AccomLikeRepository;
import org.example.hotelbooking.domain.member.entity.Member;
import org.example.hotelbooking.domain.member.repository.MemberRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.util.List;
import java.util.Optional;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class AccomLikeService {

    private final AccomLikeRepository accomLikeRepository;
    private final MemberRepository memberRepository;
    private final AccommodationRepository accommodationRepository;

    public boolean isLiked(String email, String contentId) {
        if (email == null || contentId == null) {
            return false;
        }

        Member member = memberRepository.findByEmail(email).orElse(null);
        if (member == null) {
            return false;
        }

        return accomLikeRepository.existsByMember_MemberIdAndAccommodation_ContentId(member.getMemberId(), contentId);
    }

    // 찜 토글 로직
    @Transactional
    public boolean toggleLike(String email, String contentId) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        Accommodation accommodation = accommodationRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("숙소를 찾을 수 없습니다."));

        Optional<AccomLike> existingLike = accomLikeRepository
                .findByMember_MemberIdAndAccommodation_ContentId(member.getMemberId(), contentId);

        if (existingLike.isPresent()) {
            accomLikeRepository.delete(existingLike.get());
            return false; // 찜 해제
        } else {
            AccomLike newLike = AccomLike.builder()
                    .member(member)
                    .accommodation(accommodation)
                    .build();
            accomLikeRepository.save(newLike);
            return true; // 찜 등록
        }
    }

    public List<StayResponseDto> getMyWishlist(String email, LocalDate checkIn, LocalDate checkOut, Integer guests) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        Long stayDays = 0L;
        if (checkIn != null && checkOut != null && checkOut.isAfter(checkIn)) {
            stayDays = java.time.temporal.ChronoUnit.DAYS.between(checkIn, checkOut);
        } else {
            checkIn = LocalDate.now();
            checkOut = LocalDate.now();
        }

        return accomLikeRepository.findWishlistStaysByMemberIdAndDates(member.getMemberId(), checkIn, checkOut, stayDays, guests);
    }
}

public class RoomInventory {

    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "INVENTORY_ID")
    private Long inventoryId;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "ROOM_ID", nullable = false)
    private Room room;

    @Column(name = "INVENTORY_DATE", nullable = false)

    private Integer availableCount; // 남은 객실 수

    @Builder
    public RoomInventory(Room room, LocalDate inventoryDate, Integer availableCount) {
        this.room = room;
        this.inventoryDate = inventoryDate;
        this.availableCount = availableCount;
    }

    // 💡 동시성 환경에서 호출될 재고 차감 로직
    public void decreaseCount(int count) {
        if (this.availableCount - count < 0) {
            throw new IllegalArgumentException("해당 날짜의 객실이 모두 예약되었습니다.");
        }
        this.availableCount -= count;
    }
    // 💡 예약 취소 시 재고 복구 로직
    public void increaseCount(int count) {
        this.availableCount += count;
    }

    public List<AvailableRoomResponse> getAvailableRooms(Long contentId, LocalDate date) {

                .stream()
                .map(AvailableRoomResponse::from)
                .collect(Collectors.toList());

    public List<RoomManageDto.Response> getRoomsByAccommodation(String contentId) {
        List<Room> rooms = roomRepository.findByAccommodation_ContentId(contentId);
        return rooms.stream().map(room -> {
            List<String> urls = roomImageRepository
                    .findByRoom_RoomIdOrderByDisplayOrderAsc(room.getRoomId())
                    .stream()
                    .map(RoomImage::getImageUrl)
                    .toList();
            return RoomManageDto.Response.of(room, urls);
        }).toList();
    }

    // 💡 2. 호스트: 새 객실 추가 (ROOM + ROOM_IMAGE + 향후 90일 ROOM_INVENTORY 자동 생성)
    @Transactional
    public Long createRoom(String email, String contentId, RoomManageDto.SaveRequest request) {
        Accommodation accom = validateHostOwnership(email, contentId);

        Room room = Room.builder()
                .accommodation(accom)
                .roomTitle(request.getRoomTitle())
                .roomSize(request.getRoomSize())
                .roomBaseCount(request.getRoomBaseCount())
                .roomMaxCount(request.getRoomMaxCount())
                .pricePerNight(request.getPricePerNight())
                .build();

        Room savedRoom = roomRepository.save(room);

        // 객실 이미지 저장
        saveRoomImages(savedRoom, request.getImageUrls());

        // 오늘부터 90일 치 기본 객실 재고(ROOM_INVENTORY) 자동 세팅
        int stock = (request.getInitialStock() != null && request.getInitialStock() > 0)
                ? request.getInitialStock()
                : 5;
        LocalDate today = LocalDate.now();
        List<RoomInventory> inventories = new ArrayList<>();
        for (int i = 0; i < 90; i++) {
            inventories.add(RoomInventory.builder()
                    .room(savedRoom)
                    .inventoryDate(today.plusDays(i))
                    .availableCount(stock)
                    .build());
        }
        roomInventoryRepository.saveAll(inventories);

        return savedRoom.getRoomId();
    }

    // 💡 3. 호스트: 기존 객실 정보 및 가격 수정
    @Transactional
    public void updateRoom(String email, Long roomId, RoomManageDto.SaveRequest request) {
        Room room = roomRepository.findById(roomId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 객실입니다. id=" + roomId));

        validateHostOwnership(email, room.getAccommodation().getContentId());

        room.updateRoom(
                request.getRoomTitle(),
                request.getRoomSize(),
                request.getRoomBaseCount(),
                request.getRoomMaxCount(),
                request.getPricePerNight()
        );

        // 이미지 목록이 전달된 경우 기존 이미지 교체
        if (request.getImageUrls() != null) {
            roomImageRepository.deleteByRoom_RoomId(roomId);
            saveRoomImages(room, request.getImageUrls());
        }
    }

    @Transactional
    public void deleteRoom(String email, Long roomId) {
        Room room = roomRepository.findById(roomId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 객실입니다. id=" + roomId));

        validateHostOwnership(email, room.getAccommodation().getContentId());

        roomImageRepository.deleteByRoom_RoomId(roomId);
        roomInventoryRepository.deleteByRoom_RoomId(roomId);
        roomRepository.delete(room);
    }

    private Accommodation validateHostOwnership(String email, String contentId) {
        Accommodation accom = accommodationRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 숙소입니다."));
        if (accom.getHost() == null || !accom.getHost().getEmail().equals(email)) {
            throw new IllegalArgumentException("본인 소유의 숙소 객실만 관리할 수 있습니다.");
        }
        return accom;
    }

    private void saveRoomImages(Room room, List<String> imageUrls) {
        if (imageUrls == null || imageUrls.isEmpty()) return;
        int order = 1;
        for (String url : imageUrls) {
            if (url != null && !url.isBlank()) {
                roomImageRepository.save(RoomImage.builder()
                        .room(room)
                        .imageUrl(url.trim())
                        .displayOrder(order++)
                        .build());
            }
        }
    }

public class NotificationQueryService {

    private final MemberRepository memberRepository;
    private final AccommodationRepository accommodationRepository;
    private final ReservationRepository reservationRepository;

    public List<NotificationDto> getNotifications(String email) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        List<NotificationDto> list = new ArrayList<>();
        String roleName = member.getRole().getRoleName();

        // 1. 숙소 관리자 (ROLE_HOST) 알림
        if ("ROLE_HOST".equals(roleName)) {
            // [알림 1] 숙소 등록 반려 내역
            List<Accommodation> rejectedList = accommodationRepository
                    .findByHost_MemberIdAndApprovalStatus(member.getMemberId(), AccommodationStatus.REJECTED);

            for (Accommodation a : rejectedList) {
                String reason = a.getRejectReason() != null ? a.getRejectReason() : "사유가 입력되지 않았습니다.";
                list.add(NotificationDto.builder()
                        .id("REJECT_" + a.getContentId())
                        .title("숙소 신청이 반려되었습니다.")
                        .message("숙소: " + a.getTitle() + " / 반려 사유 : " + reason)
                        .type("ACCOM_REJECT")
                        .createdAt(a.getCreatedAt())
                        .build());
            }

            // [알림 2] 숙소 예약 신청 알림 (추후 예약 신청 연동용)
            List<Reservation> hostReservations = reservationRepository.findReservationsByHostId(member.getMemberId());
            for (Reservation r : hostReservations) {
                // 최근 예약 알림 (예시: COMPLETED/RESERVED 상태)
                list.add(NotificationDto.builder()
                        .id("RES_" + r.getReservationId())
                        .title("새로운 숙소 예약이 접수되었습니다.")
                        .message("예약자: " + r.getMember().getName() + " (" + r.getCheckInDate() + " ~ " + r.getCheckOutDate() + ")")
                        .type("RESERVATION_NEW")
                        .createdAt(r.getCreatedAt())
                        .build());
            }
        }

        // 2. 총괄 관리자 (ROLE_ADMIN) 알림
        if ("ROLE_ADMIN".equals(roleName)) {
            // [알림 3] 신규 등록되어 심사 대기 중인 숙소 알림
            List<Accommodation> pendingList = accommodationRepository
                    .findByApprovalStatusOrderByCreatedAtDesc(AccommodationStatus.PENDING);

            for (Accommodation a : pendingList) {
                list.add(NotificationDto.builder()
                        .id("PENDING_" + a.getContentId())
                        .title("신규 숙소 승인 요청이 있습니다.")
                        .message("숙소명: " + a.getTitle() + " (호스트: " + a.getHost().getName() + ")")
                        .type("ACCOM_PENDING")
                        .createdAt(a.getCreatedAt())
                        .build());
            }
        }

        // 최신 등록 순 정렬
        list.sort((o1, o2) -> {
            if (o1.getCreatedAt() == null || o2.getCreatedAt() == null) return 0;
            return o2.getCreatedAt().compareTo(o1.getCreatedAt());
        });

        return list;
    }

        Map<String, Object> params = new HashMap<>();
        params.put("paymentKey", paymentKey);
        params.put("orderId", orderId);
        params.put("amount", amount);

        HttpEntity<Map<String, Object>> request = new HttpEntity<>(params, getHeaders());

            restTemplate.postForEntity(url, request, String.class);

        } catch (Exception e) {
            log.error("❌ 토스 결제 승인 실패 (통신 오류): {}", e.getMessage());
            throw new RuntimeException("결제 승인 중 통신 오류가 발생했습니다.");

        org.example.hotelbooking.domain.member.entity.Member member = memberRepository.findByEmail(email)

    public void verifyResetCode(String email, String inputCode) {

        String savedCode = redisTemplate.opsForValue().get(redisKey);

        if (savedCode == null || !savedCode.equals(inputCode)) {

        }

        // 인증 성공 시, 기존 인증 코드는 삭제

        // "이메일 인증 완료" 상태를 Redis에 10분간 저장 (합격 티켓)

        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        // 이름 변경
        if (request.getName() != null && !request.getName().isBlank()) {
            member.updateInfo(request.getName().trim());
        }

        // 비밀번호 변경 (입력된 경우에만 검증 후 변경)

                throw new IllegalArgumentException("소셜 로그인 회원은 비밀번호를 변경할 수 없습니다.");
            }

            if (request.getCurrentPassword() == null || !passwordEncoder.matches(request.getCurrentPassword(), member.getPassword())) {
                throw new IllegalArgumentException("현재 비밀번호가 일치하지 않습니다.");
            }

            member.updatePassword(passwordEncoder.encode(request.getNewPassword()));
        }

        return jwtTokenProvider.createToken(member.getEmail(), member.getRole().getRoleName(), member.getName());

    @Transactional
    public String downgradeHost(String email) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        if (!"ROLE_HOST".equals(member.getRole().getRoleName())) {
            throw new IllegalArgumentException("숙소 관리자 권한을 가진 회원만 신청할 수 있습니다.");
        }

        // [검증 1] 내 숙소에 투숙 예정/진행 중인 예약이 남아있는지 확인
        LocalDate today = LocalDate.now();
        List<Reservation> hostReservations = reservationRepository.findReservationsByHostId(member.getMemberId());
        boolean hasUpcomingReservation = hostReservations.stream()
                .anyMatch(r -> !"CANCELED".equals(r.getStatus()) && r.getCheckOutDate().isAfter(today));

        if (hasUpcomingReservation) {
            throw new IllegalStateException("현재 투숙 예정이거나 진행 중인 예약이 존재하여 숙소 관리자 자격을 해지할 수 없습니다.");

            );
        }

        Role userRole = roleRepository.findByRoleName("ROLE_USER")
                .orElseThrow(() -> new IllegalArgumentException("ROLE_USER 권한이 존재하지 않습니다."));

        member.downgradeToUser(userRole);

        return jwtTokenProvider.createToken(member.getEmail(), userRole.getRoleName(), member.getName());

    @Transactional
    public void withdrawMember(String email) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        LocalDate today = LocalDate.now();

        // [검증 1] 일반 유저(게스트)로서 체크아웃 전인 진행 예정 예약이 있는지 확인
        List<Reservation> myGuestReservations = reservationRepository
                .findByMember_MemberIdOrderByCreatedAtDesc(member.getMemberId());
        boolean hasUpcomingGuestRes = myGuestReservations.stream()
                .anyMatch(r -> !"CANCELED".equals(r.getStatus()) && r.getCheckOutDate().isAfter(today));

        if (hasUpcomingGuestRes) {
            throw new IllegalStateException("이용 예정이거나 투숙 중인 예약이 존재하여 탈퇴할 수 없습니다. 예약을 먼저 취소해 주세요.");
        }

        // [검증 2] 호스트(ROLE_HOST)인 경우 보유 숙소나 호스트 예약이 남아있는지 확인
        if ("ROLE_HOST".equals(member.getRole().getRoleName())) {
            List<Reservation> hostReservations = reservationRepository.findReservationsByHostId(member.getMemberId());
            boolean hasUpcomingHostRes = hostReservations.stream()
                    .anyMatch(r -> !"CANCELED".equals(r.getStatus()) && r.getCheckOutDate().isAfter(today));

            List<Accommodation> myAccommodations = accommodationRepository.findByHost_Email(email);

            if (hasUpcomingHostRes || !myAccommodations.isEmpty()) {
                throw new IllegalStateException("보유 중인 숙소 또는 진행 예정인 예약이 존재하여 탈퇴할 수 없습니다. 숙소 삭제 후 다시 시도해 주세요.");
            }
        }

        Role userRole = roleRepository.findByRoleName("ROLE_USER").orElse(member.getRole());

        // 상태 WITHDRAWN 변경 및 이메일/이름 Unknown 처리 (기존 예약·리뷰 FK는 깨지지 않고 유지됨!)
        member.withdraw(userRole);
    }

    @Transactional
    public String kakaoLogin(String code) {
        // 1. 카카오 서버로 인가 코드를 보내서 '액세스 토큰'을 받아옵니다.
        String accessToken = getAccessToken(code);

        // 2. 받아온 액세스 토큰으로 카카오 유저 정보(고유번호, 닉네임)를 가져옵니다.
        JsonNode userInfo = getUserInfo(accessToken);
        String kakaoId = userInfo.get("id").asText();
        String nickname = userInfo.get("properties").get("nickname").asText();

        // 3. 이메일을 못 받으므로, 카카오 고유번호를 이용해 가짜 이메일을 생성합니다.
        String dummyEmail = "kakao_" + kakaoId + "@hotel.com";

        // 4. DB에 이미 있는 유저인지 확인하고, 없으면 자동 회원가입 시킵니다.
        Member member = memberRepository.findByEmail(dummyEmail).orElseGet(() -> {
            Role userRole = roleRepository.findByRoleName("ROLE_USER")
                    .orElseThrow(() -> new IllegalArgumentException("기본 권한이 없습니다."));

            // 비밀번호는 랜덤 UUID로 생성해서 암호화 (어차피 카카오 로그인은 비밀번호를 안 씁니다)
            Member newMember = Member.builder()
                    .email(dummyEmail)
                    .password(passwordEncoder.encode(UUID.randomUUID().toString()))
                    .name(nickname)
                    .role(userRole)
                    .build();
            return memberRepository.save(newMember);
        });

        // 5. 회원가입(또는 로그인)이 완료되었으니, 우리 사이트 전용 JWT 토큰을 발급합니다.
        return jwtTokenProvider.createToken(member.getEmail(), member.getRole().getRoleName(), member.getName());
    }

    private String getAccessToken(String code) {
        RestTemplate rt = new RestTemplate();
        HttpHeaders headers = new HttpHeaders();
        headers.add("Content-type", "application/x-www-form-urlencoded;charset=utf-8");

        MultiValueMap<String, String> params = new LinkedMultiValueMap<>();
        params.add("grant_type", "authorization_code");
        params.add("client_id", clientId);
        params.add("redirect_uri", redirectUri);
        params.add("code", code);

        HttpEntity<MultiValueMap<String, String>> request = new HttpEntity<>(params, headers);
        ResponseEntity<String> response = rt.postForEntity("https://kauth.kakao.com/oauth/token", request, String.class);

        try {
            ObjectMapper mapper = new ObjectMapper();
            JsonNode node = mapper.readTree(response.getBody());
            return node.get("access_token").asText();
        } catch (Exception e) {
            throw new RuntimeException("카카오 토큰 발급 중 오류가 발생했습니다.", e);
        }
    }

    private JsonNode getUserInfo(String accessToken) {
        RestTemplate rt = new RestTemplate();
        HttpHeaders headers = new HttpHeaders();
        headers.add("Authorization", "Bearer " + accessToken);
        headers.add("Content-type", "application/x-www-form-urlencoded;charset=utf-8");

        HttpEntity<MultiValueMap<String, String>> request = new HttpEntity<>(headers);
        ResponseEntity<String> response = rt.exchange("https://kapi.kakao.com/v2/user/me", HttpMethod.GET, request, String.class);

        try {
            ObjectMapper mapper = new ObjectMapper();
            return mapper.readTree(response.getBody());
        } catch (Exception e) {
            throw new RuntimeException("카카오 유저 정보 조회 중 오류가 발생했습니다.", e);
        }
    }

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.example.hotelbooking.domain.reservation.repository.ReservationRepository;
import org.example.hotelbooking.domain.reservation.service.ReservationService;
import org.redisson.api.RLock;
import org.redisson.api.RedissonClient;
import org.springframework.stereotype.Component;

import java.time.LocalDate;
import java.util.concurrent.TimeUnit;

@Slf4j
@Component
@RequiredArgsConstructor
public class ReservationLockFacade {

    private final RedissonClient redissonClient;
    private final ReservationService reservationService;
    private final ReservationRepository reservationRepository;

    public void reserveRoomWithLock(Long memberId, Long roomId, LocalDate checkIn, LocalDate checkOut, Integer totalPrice) {
        // 객실 고유 ID를 기반으로 락 키 생성 (해당 객실에 대해서만 줄을 세움)
        String lockKey = "room_lock:" + roomId;
        RLock lock = redissonClient.getLock(lockKey);

        try {
            // 락 획득 시도: 최대 5초 대기, 락 획득 후 3초간 유지
            boolean isLocked = lock.tryLock(5, 3, TimeUnit.SECONDS);

            if (!isLocked) {
                log.warn("락 획득 실패 - roomId: {}", roomId);
                throw new RuntimeException("현재 예약자가 많아 처리가 지연되고 있습니다. 잠시 후 다시 시도해주세요.");
            }

            // 💡 락 획득 성공 시, 안전하게 트랜잭션(예약 및 재고 차감) 실행
            reservationService.createReservation(memberId, roomId, checkIn, checkOut, totalPrice);

        } catch (InterruptedException e) {
            log.error("예약 락 대기 중 인터럽트 발생", e);
            throw new RuntimeException("예약 처리 중 시스템 오류가 발생했습니다.");
        } finally {
            // 로직이 끝나면 반드시 락 반납
            if (lock.isLocked() && lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }

            throw new RuntimeException("취소 처리 중 시스템 오류가 발생했습니다.");

    @Transactional
    public void createReservation(Long memberId, Long roomId, LocalDate checkIn, LocalDate checkOut, Integer totalPrice) {
        // 1. 체크인 ~ 체크아웃 전날(-1일)까지의 재고 조회
        List<RoomInventory> inventories = roomInventoryRepository
                .findInventoriesForReservation(roomId, checkIn, checkOut.minusDays(1));

        // 2. 요청한 숙박 일수와 DB에 세팅된 재고 일수가 일치하는지 검증
        long days = java.time.temporal.ChronoUnit.DAYS.between(checkIn, checkOut);
        if (inventories.size() != days) {
            throw new IllegalArgumentException("예약할 수 없는 날짜가 포함되어 있습니다.");
        }

        // 3. 재고 차감 (Dirty Checking으로 자동 UPDATE 쿼리 발생)
        for (RoomInventory inventory : inventories) {
            inventory.decreaseCount(1);
        }

        // 4. 예약 정보 저장
        Member member = memberRepository.findById(memberId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 회원입니다."));
        Room room = roomRepository.findById(roomId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 객실입니다."));

        Reservation reservation = Reservation.builder()
                .member(member)
                .room(room)
                .checkInDate(checkIn)
                .checkOutDate(checkOut)
                .totalPrice(totalPrice)
                .status("COMPLETED")
                .build();

        Reservation savedReservation = reservationRepository.save(reservation);

        // 💡 5. 예약 확정 안내 이메일 비동기 발송
        Accommodation accom = room.getAccommodation();
        mailService.sendReservationConfirmation(
                member.getEmail(),
                member.getName(),
                savedReservation.getReservationId(),
                accom.getContentId(),
                accom.getTitle(),
                accom.getAddr(),
                accom.getTel(),
                accom.getCheckinTime(),
                accom.getCheckoutTime(),
                room.getRoomTitle(),
                checkIn,
                checkOut,
                totalPrice
        );
    }

    @Transactional
    public void cancelReservation(Long reservationId, Long requesterMemberId) {
        // 1. 예약 정보 조회
        Reservation reservation = reservationRepository.findById(reservationId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 예약입니다."));

        Long guestMemberId = reservation.getMember().getMemberId();
        Member host = reservation.getRoom().getAccommodation().getHost();
        Long hostMemberId = (host != null) ? host.getMemberId() : null;

        boolean isGuest = guestMemberId.equals(requesterMemberId);
        boolean isHostOwner = hostMemberId != null && hostMemberId.equals(requesterMemberId);

        // 2. 본인의 예약인지 검증 및 취소, 숙소 관리자 예약 취소
        if (!isGuest && !isHostOwner) {
            throw new IllegalArgumentException("예약자 본인 또는 해당 숙소의 관리자만 예약을 취소할 수 있습니다.");
        }

        // 3. 예약 상태 취소로 변경 (Dirty Checking)
        reservation.cancel();

        // 4. 재고 복구
        List<RoomInventory> inventories = roomInventoryRepository.findInventoriesForReservation(
                reservation.getRoom().getRoomId(),
                reservation.getCheckInDate(),
                reservation.getCheckOutDate().minusDays(1)
        );

        for (RoomInventory inventory : inventories) {
            inventory.increaseCount(1);
        }
    }

                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 예약입니다."));

    private final StringRedisTemplate redisTemplate;
    private final TourAttractionRepository tourRepository;
    private final AccommodationRepository accommodationRepository;

    public List<NearbyTourResponseDto> getNearbyTours(String accomContentId) {
        // 기준점(해당 숙소)의 좌표를 오라클에서 가져옵니다.
        Accommodation accom = accommodationRepository.findById(accomContentId)
                .orElseThrow(() -> new IllegalArgumentException("숙소를 찾을 수 없습니다."));

        double accomX = Double.parseDouble(accom.getMapX());
        double accomY = Double.parseDouble(accom.getMapY());

        // 2. Redis에 반경 20km 이내의 관광지를 거리순(오름차순)으로 요청합니다.
        String key = "tour:geo:locations";
        Circle circle = new Circle(new Point(accomX, accomY), new Distance(20.0, Metrics.KILOMETERS));

        RedisGeoCommands.GeoRadiusCommandArgs args =
                RedisGeoCommands.GeoRadiusCommandArgs.newGeoRadiusArgs()
                        .includeDistance()
                        .sortAscending()
                        .limit(9); // 최대 9개까지만 노출

        GeoResults<RedisGeoCommands.GeoLocation<String>> geoResults =
                redisTemplate.opsForGeo().radius(key, circle, args);

        List<NearbyTourResponseDto> resultList = new ArrayList<>();

        if (geoResults != null) {
            for (GeoResult<RedisGeoCommands.GeoLocation<String>> geoResult : geoResults) {
                // Redis에서 가져온 관광지 ID와 거리
                String tourId = geoResult.getContent().getName();
                double distance = geoResult.getDistance().getValue();

                // 관광지 ID로 오라클 DB에서 이름, 사진 등을 가져와 DTO로 합칩니다.
                tourRepository.findById(tourId).ifPresent(tour -> {
                    double formattedDistance = Math.round(distance * 10) / 10.0;

                    resultList.add(new NearbyTourResponseDto(
                            tour.getContentId(),
                            tour.getTitle(),
                            tour.getFirstImage(),
                            formattedDistance
                    ));
                });
            }
        }
        return resultList;
    }

@Transactional(readOnly = true)
public class TourService {

    private final TourAttractionRepository tourAttractionRepository;
    private final TourImageRepository tourImageRepository;
    private final TourLikeRepository tourLikeRepository;
    private final MemberRepository memberRepository;

    // 1. 전체 관광지 목록 조회 (지도 표시용)
    public List<TourDto.SummaryResponse> getAllTours() {
        return tourAttractionRepository.findAllByOrderByTitleAsc()
                .stream()
                .map(TourDto.SummaryResponse::from)
                .toList();
    }

    public TourDto.DetailResponse getTourDetail(String contentId) {
        TourAttraction tour = tourAttractionRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("관광지 정보를 찾을 수 없습니다. contentId=" + contentId));

        List<TourImage> images = tourImageRepository.findByTourAttraction_ContentIdOrderByImageIdAsc(contentId);
        return TourDto.DetailResponse.of(tour, images);
    }

    // 3. 관광지 찜 여부 확인
    public boolean isLiked(String email, String contentId) {
        if (email == null || contentId == null) return false;
        Member member = memberRepository.findByEmail(email).orElse(null);
        if (member == null) return false;
        return tourLikeRepository.existsByMember_MemberIdAndTourAttraction_ContentId(member.getMemberId(), contentId);
    }

    // 4. 관광지 찜 토글
    @Transactional
    public boolean toggleTourLike(String email, String contentId) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        TourAttraction tour = tourAttractionRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("관광지를 찾을 수 없습니다."));

        Optional<TourLike> existing = tourLikeRepository
                .findByMember_MemberIdAndTourAttraction_ContentId(member.getMemberId(), contentId);

        if (existing.isPresent()) {
            tourLikeRepository.delete(existing.get());
            return false;
        } else {
            tourLikeRepository.save(TourLike.builder()
                    .member(member)
                    .tourAttraction(tour)
                    .build());
            return true;
        }
    }

                .stream()
                .map(tl -> TourDto.SummaryResponse.from(tl.getTourAttraction()))
                .toList();

import java.time.LocalDate;
import java.time.temporal.ChronoUnit;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class StayService {

    private final AccommodationRepository accommodationRepository;
    private final RoomRepository roomRepository;
    private final RoomImageRepository roomImageRepository;
    private final RoomInventoryRepository roomInventoryRepository; // 💡 재고 조회용 주입

    public List<StayResponseDto> getStays(String region, String type, String query, LocalDate checkIn, LocalDate checkOut, Integer guests) {
        Long stayDays = 0L;

        if (checkIn != null && checkOut != null && checkOut.isAfter(checkIn)) {
            stayDays = ChronoUnit.DAYS.between(checkIn, checkOut);
        } else {
            checkIn = LocalDate.now();
            checkOut = LocalDate.now();
        }

        return accommodationRepository.searchStays(region, type, query, checkIn, checkOut, stayDays, guests);
    }

    // 💡 [수정] 체크인/체크아웃 날짜를 받아 객실별 잔여 개수 계산
    public StayDetailResponseDto getStayDetail(String contentId, LocalDate checkIn, LocalDate checkOut) {
        Accommodation accommodation = accommodationRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("숙소를 찾을 수 없습니다. contentId=" + contentId));

        // 날짜가 지정되지 않았으면 기본 오늘 체크인 ~ 내일 체크아웃(1박) 기준
        if (checkIn == null) {
            checkIn = LocalDate.now();
        }
        if (checkOut == null || !checkOut.isAfter(checkIn)) {
            checkOut = checkIn.plusDays(1);
        }

        long stayDays = ChronoUnit.DAYS.between(checkIn, checkOut);

        List<Room> rooms = roomRepository.findByAccommodation_ContentId(contentId);

        // 해당 숙소의 기간 내 모든 재고 조회 후 roomId별로 그룹핑
        List<RoomInventory> inventories = roomInventoryRepository.findByContentIdAndDateRange(contentId, checkIn, checkOut);
        Map<Long, List<RoomInventory>> invByRoomMap = inventories.stream()
                .collect(Collectors.groupingBy(ri -> ri.getRoom().getRoomId()));

        Map<Long, List<RoomImage>> roomImagesMap = new HashMap<>();
        Map<Long, Integer> roomAvailableMap = new HashMap<>();

        for (Room room : rooms) {
            List<RoomImage> images = roomImageRepository.findByRoom_RoomIdOrderByDisplayOrderAsc(room.getRoomId());
            roomImagesMap.put(room.getRoomId(), images);

            List<RoomInventory> roomInvs = invByRoomMap.getOrDefault(room.getRoomId(), List.of());
            // 숙박 일수(stayDays)만큼 모든 날짜에 재고 데이터가 있어야 하며, 그중 최소 재고를 잔여 객실 수로 설정
            if (roomInvs.size() < stayDays) {
                roomAvailableMap.put(room.getRoomId(), 0);
            } else {
                int minAvailable = roomInvs.stream()
                        .mapToInt(RoomInventory::getAvailableCount)
                        .min()
                        .orElse(0);
                roomAvailableMap.put(room.getRoomId(), Math.max(minAvailable, 0));
            }
        }

        return StayDetailResponseDto.of(accommodation, rooms, roomImagesMap, roomAvailableMap);
    }

                // JWT를 사용하므로 서버 세션을 사용하지 않음
                .sessionManagement(session ->
                        session.sessionCreationPolicy(SessionCreationPolicy.STATELESS)
                )

                // API 접근 권한 설정
                .authorizeHttpRequests(auth -> auth
                        // 💡 스프링 부트 기본 에러 페이지 포워딩 시 403 마스킹 방지
                        .requestMatchers("/error").permitAll()

                        // 로그인 / 회원가입
                        .requestMatchers("/api/v1/auth/**").permitAll()

                        // 숙소 조회
                        .requestMatchers("/api/v1/accommodations/**").permitAll()

                        // 숙소 조회 API
                        .requestMatchers("/api/v1/stays/**").permitAll()

                        // 관광지 조회 api
                        .requestMatchers("/api/v1/tours/**").permitAll()

                        // 비밀번호 찾기 위한 허용 설정
                        .requestMatchers("/api/v1/members/password/**").permitAll()

                        // 토스페이먼츠 클라이언트 키 조회 허용
                        .requestMatchers("/api/v1/payments/client-key").permitAll()

                        // 숙소 리뷰 조회 허용
                        .requestMatchers(org.springframework.http.HttpMethod.GET, "/api/v1/reviews/stays/**").permitAll()

                        // 그 외 요청은 JWT 인증 필요
                        .anyRequest().authenticated()
                )

                // JWT 인증 필터 등록
                .addFilterBefore(
                        new JwtAuthenticationFilter(jwtTokenProvider),
                        UsernamePasswordAuthenticationFilter.class
                );

        return http.build();
    }

package org.example.hotelbooking.global.security;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.util.StringUtils;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.Collections;

// 💡 롬복을 써서 JwtTokenProvider를 자동으로 주입받습니다.
@RequiredArgsConstructor
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private final JwtTokenProvider jwtTokenProvider;

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
            throws ServletException, IOException {

        // 1. 사용자 요청 헤더에서 JWT 토큰을 쓱 뽑아옵니다. (아래 만들어둔 resolveToken 메서드 사용)
        String token = resolveToken(request);

        // 2. 토큰이 비어있지 않고, 진품(유효기간 안 지남 등)인지 검사합니다.
        if (token != null && jwtTokenProvider.validateToken(token)) {

            // 3. 진품이라면 토큰 안에 적혀있는 사용자 이메일(Subject)과 역할(Role)을 꺼냅니다.
            String email = jwtTokenProvider.getUserEmail(token);
            String role = jwtTokenProvider.getUserRole(token);

            // 4. 꺼낸 정보를 바탕으로 "이 사람은 인증된 사용자입니다" 라는 확인증(Authentication 객체)을 만듭니다.
            Authentication authentication = new UsernamePasswordAuthenticationToken(
                    email,
                    null,
                    Collections.singletonList(new SimpleGrantedAuthority(role))
            );

            // 5. 스프링 시큐리티의 핵심 저장소(SecurityContext)에 이 확인증을 예쁘게 보관해 둡니다.
            //    (이렇게 해두면 나중에 컨트롤러에서 @AuthenticationPrincipal 로 이 사람 정보를 꺼내 쓸 수 있습니다!)
            SecurityContextHolder.getContext().setAuthentication(authentication);
        }

        // 6. 검사가 끝났으니 다음 필터나 컨트롤러로 무사히 넘겨줍니다.
        filterChain.doFilter(request, response);
    }

    // 💡 프론트엔드가 보낸 요청(Header) 주머니에서 "Bearer xxxx.yyyy.zzzz" 형태의 토큰만 쏙 빼내는 기능
    private String resolveToken(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        // 토큰이 존재하고, "Bearer "로 시작하면 (7글자)
        if (StringUtils.hasText(bearerToken) && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7); // "Bearer " 이후의 진짜 토큰 부분만 잘라서 반환
        }
        return null; // 토큰이 없으면 null 반환
    }
}

package org.example.hotelbooking.global.security;

import io.jsonwebtoken.*;
import io.jsonwebtoken.io.Decoders;
import io.jsonwebtoken.security.Keys;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.security.Key;
import java.util.Date;

@Slf4j
@Component
public class JwtTokenProvider {

    private final Key key;
    private final long expirationTime;

    // yml 또는 properties 파일의 값을 불러와서 Key 객체로 만듭니다.
    public JwtTokenProvider(@Value("${jwt.secret}") String secretKey,
                            @Value("${jwt.expiration}") long expirationTime) {
        byte[] keyBytes = Decoders.BASE64.decode(secretKey);
        this.key = Keys.hmacShaKeyFor(keyBytes);
        this.expirationTime = expirationTime;
    }

    // 💡 1. 토큰 생성 (로그인 성공 시 프론트엔드로 내려줄 출입증)
    // 프론트엔드에서 로그인 시 실제 닉네임을 띄우기 위해 String name, .claim("name", name) 추가
    public String createToken(String email, String role, String name) {
        Date now = new Date();
        Date validity = new Date(now.getTime() + this.expirationTime);

        return Jwts.builder()
                .setSubject(email) // 토큰 주인을 이메일로 기록
                .claim("role", role) // 권한 정보 기록 (예: ROLE_USER)
                .claim("name", name) // 💡 토큰 claim에 DB name 추가
                .setIssuedAt(now) // 발행 시간
                .setExpiration(validity) // 만료 시간
                .signWith(key, SignatureAlgorithm.HS256) // 암호화 알고리즘
                .compact();
    }

    // 💡 2. 토큰 유효성 검증 (API 요청이 올 때마다 진짜 출입증인지 확인)
    public boolean validateToken(String token) {
        try {
            Jwts.parserBuilder().setSigningKey(key).build().parseClaimsJws(token);
            return true;
        } catch (io.jsonwebtoken.security.SecurityException | MalformedJwtException e) {
            log.info("잘못된 JWT 서명입니다.");
        } catch (ExpiredJwtException e) {
            log.info("만료된 JWT 토큰입니다.");
        } catch (UnsupportedJwtException e) {
            log.info("지원되지 않는 JWT 토큰입니다.");
        } catch (IllegalArgumentException e) {
            log.info("JWT 토큰이 잘못되었습니다.");
        }
        return false;

    }
    // 💡 3. 토큰을 분해해서 주인의 이메일(Subject)을 꺼내는 기능
    public String getUserEmail(String token) {
        return Jwts.parserBuilder().setSigningKey(key).build()
                .parseClaimsJws(token)
                .getBody()
                .getSubject();
    }

    // 💡 4. 토큰을 분해해서 주인의 권한(Role)을 꺼내는 기능
    public String getUserRole(String token) {
        return Jwts.parserBuilder().setSigningKey(key).build()
                .parseClaimsJws(token)
                .getBody()
                .get("role", String.class);
    }
}
