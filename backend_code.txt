package org.example.hotelbooking.config;

import org.redisson.Redisson;
import org.redisson.api.RedissonClient;
import org.redisson.config.Config;
import org.springframework.cache.CacheManager;
import org.springframework.cache.annotation.EnableCaching;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.data.redis.cache.RedisCacheConfiguration;
import org.springframework.data.redis.cache.RedisCacheManager;
import org.springframework.data.redis.connection.RedisConnectionFactory;
import org.springframework.data.redis.serializer.GenericJackson2JsonRedisSerializer;
import org.springframework.data.redis.serializer.RedisSerializationContext;
import org.springframework.data.redis.serializer.StringRedisSerializer;

import java.time.Duration;

@Configuration
@EnableCaching // 💡 스프링 캐싱 기능을 활성화합니다.
public class RedissonConfig {

    private static final String REDIS_URL = "redis://127.0.0.1:6379";

    // Redisson 빈 (동시성 제어 / 분산 락 용도)
    @Bean
    public RedissonClient redissonClient() {
        Config config = new Config();
        config.useSingleServer().setAddress(REDIS_URL);
        return Redisson.create(config);
    }

    // CacheManager 빈 (조회 성능 최적화 / Redis 캐싱 용도)
    @Bean
    public CacheManager cacheManager(RedisConnectionFactory redisConnectionFactory) {
        RedisCacheConfiguration redisCacheConfiguration = RedisCacheConfiguration.defaultCacheConfig()
                // Key를 사람이 읽을 수 있는 문자열(String)로 저장
                .serializeKeysWith(RedisSerializationContext.SerializationPair.fromSerializer(new StringRedisSerializer()))
                // Value를 DTO 객체에서 JSON 형태로 변환하여 저장
                .serializeValuesWith(RedisSerializationContext.SerializationPair.fromSerializer(new GenericJackson2JsonRedisSerializer()))
                // 캐시 데이터의 유효 시간(TTL)을 10분으로 설정
                .entryTtl(Duration.ofMinutes(10));

        return RedisCacheManager.RedisCacheManagerBuilder
                .fromConnectionFactory(redisConnectionFactory)
                .cacheDefaults(redisCacheConfiguration)
                .build();
    }
}

package org.example.hotelbooking.domain.like.repository;

import org.example.hotelbooking.domain.accommodation.dto.StayResponseDto;
import org.example.hotelbooking.domain.like.entity.AccomLike;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

import java.time.LocalDate;
import java.util.List;
import java.util.Optional;

public interface AccomLikeRepository extends JpaRepository<AccomLike, Long> {
    boolean existsByMember_MemberIdAndAccommodation_ContentId(Long memberId, String contentId);

    Optional<AccomLike> findByMember_MemberIdAndAccommodation_ContentId(Long memberId, String contentId);

    List<AccomLike> findByMember_MemberIdOrderByCreatedAtDesc(Long memberId);

    // 💡 날짜가 있으면 재고가 있는 찜 숙소만 필터링, 날짜가 없으면(CAST IS NULL) 전체 찜 숙소 반환
    @Query("SELECT new org.example.hotelbooking.domain.accommodation.dto.StayResponseDto(" +
            "a.contentId, a.host.memberId, a.title, a.lodgingType, a.tel, a.zipcode, a.addr, " +
            "a.checkinTime, a.checkoutTime, a.roomCount, a.roomType, a.parking, a.subFacility, " +
            "a.overview, a.firstImage, a.firstImage2, a.mapX, a.mapY, " +
            "COALESCE((SELECT MIN(r.pricePerNight) FROM Room r WHERE r.accommodation = a), 0), " +
            "COALESCE((SELECT SUM(ri.availableCount) FROM RoomInventory ri JOIN ri.room r " +
            "          WHERE r.accommodation = a " +
            "            AND ri.inventoryDate = :checkIn " +
            "            AND (:guests IS NULL OR :guests <= 0 OR CAST(COALESCE(r.roomMaxCount, '0') AS integer) >= :guests)), 0L)) " +
            "FROM AccomLike al " +
            "JOIN al.accommodation a " +
            "WHERE al.member.memberId = :memberId " +
            // 💡 1. 인원 조건
            "  AND (:guests IS NULL OR :guests <= 0 OR EXISTS (" +
            "       SELECT 1 FROM Room r " +
            "       WHERE r.accommodation = a " +
            "         AND CAST(COALESCE(r.roomMaxCount, '0') AS integer) >= :guests)) " +
            // 💡 2. 날짜 조건 (:stayDays = 0L 이면 전체 찜 숙소 그대로 반환!)
            "  AND (:stayDays = 0L OR EXISTS (" +
            "       SELECT 1 FROM Room r " +
            "       WHERE r.accommodation = a " +
            "         AND (:guests IS NULL OR :guests <= 0 OR CAST(COALESCE(r.roomMaxCount, '0') AS integer) >= :guests) " +
            "         AND (SELECT COUNT(ri) FROM RoomInventory ri " +
            "              WHERE ri.room = r " +
            "                AND ri.inventoryDate >= :checkIn " +
            "                AND ri.inventoryDate < :checkOut " +
            "                AND ri.availableCount > 0) = :stayDays)) " +
            "ORDER BY al.createdAt DESC")
    List<StayResponseDto> findWishlistStaysByMemberIdAndDates(
            @Param("memberId") Long memberId,
            @Param("checkIn") LocalDate checkIn,
            @Param("checkOut") LocalDate checkOut,
            @Param("stayDays") Long stayDays,
            @Param("guests") Integer guests
    );

    // 💡 숙소 삭제 시 해당 숙소를 찜한 내역 일괄 삭제
    void deleteByAccommodation_ContentId(String contentId);
}

package org.example.hotelbooking.domain.like.entity;

import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.member.entity.Member;
import jakarta.persistence.*;
import lombok.*;
import java.time.LocalDateTime;

@Entity
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@Table(name = "ACCOM_LIKE")
public class AccomLike {

    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "LIKE_ID")
    private Long likeId;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "MEMBER_ID", nullable = false)
    private Member member;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "CONTENT_ID", nullable = false)
    private Accommodation accommodation;

    @Column(name = "CREATED_AT", updatable = false)
    private LocalDateTime createdAt;

    @PrePersist
    protected void onCreate() {
        this.createdAt = LocalDateTime.now();
    }
    @Builder
    public AccomLike(Member member, Accommodation accommodation) {
        this.member = member;
        this.accommodation = accommodation;
    }
}

import org.springframework.format.annotation.DateTimeFormat;

import org.springframework.security.core.annotation.AuthenticationPrincipal;

import java.util.List;
import java.util.Map;

@RestController
@RequestMapping("/api/v1/likes/accommodations")
@RequiredArgsConstructor
public class AccomLikeController {

    private final AccomLikeService accomLikeService;

    // 1. 찜 토글
    @PostMapping("/{contentId}")
    public ResponseEntity<Map<String, Object>> toggleLike(
            @PathVariable String contentId,
            @AuthenticationPrincipal String email
    ) {
        boolean liked = accomLikeService.toggleLike(email, contentId);
        return ResponseEntity.ok(Map.of(
                "liked", liked,
                "message", liked ? "위시리스트에 저장되었습니다." : "위시리스트에서 제외되었습니다."
        ));
    }

    // 2. 내 위시리스트 조회 (날짜 및 인원 조건 반영)
    @GetMapping("/my")
    public ResponseEntity<List<StayResponseDto>> getMyWishlist(
            @AuthenticationPrincipal String email,
            @RequestParam(value = "checkIn", required = false) @DateTimeFormat(pattern = "yyyy-MM-dd") LocalDate checkIn,
            @RequestParam(value = "checkOut", required = false) @DateTimeFormat(pattern = "yyyy-MM-dd") LocalDate checkOut,
            @RequestParam(value = "guests", required = false) Integer guests
    ) {
        return ResponseEntity.ok(accomLikeService.getMyWishlist(email, checkIn, checkOut, guests));
    }

    // 3. 특정 숙소 찜 여부 단건 확인
    @GetMapping("/{contentId}/status")
    public ResponseEntity<Boolean> getLikeStatus(
            @PathVariable String contentId,
            @AuthenticationPrincipal String email
    ) {
        return ResponseEntity.ok(accomLikeService.isLiked(email, contentId));
    }
}

package org.example.hotelbooking.domain.like.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.dto.StayResponseDto;
import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.accommodation.repository.AccommodationRepository;
import org.example.hotelbooking.domain.like.entity.AccomLike;
import org.example.hotelbooking.domain.like.repository.AccomLikeRepository;
import org.example.hotelbooking.domain.member.entity.Member;
import org.example.hotelbooking.domain.member.repository.MemberRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.util.List;
import java.util.Optional;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class AccomLikeService {

    private final AccomLikeRepository accomLikeRepository;
    private final MemberRepository memberRepository;
    private final AccommodationRepository accommodationRepository;

    public boolean isLiked(String email, String contentId) {
        if (email == null || contentId == null) {
            return false;
        }

        Member member = memberRepository.findByEmail(email).orElse(null);
        if (member == null) {
            return false;
        }

        return accomLikeRepository.existsByMember_MemberIdAndAccommodation_ContentId(member.getMemberId(), contentId);
    }

    // 찜 토글 로직
    @Transactional
    public boolean toggleLike(String email, String contentId) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        Accommodation accommodation = accommodationRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("숙소를 찾을 수 없습니다."));

        Optional<AccomLike> existingLike = accomLikeRepository
                .findByMember_MemberIdAndAccommodation_ContentId(member.getMemberId(), contentId);

        if (existingLike.isPresent()) {
            accomLikeRepository.delete(existingLike.get());
            return false; // 찜 해제
        } else {
            AccomLike newLike = AccomLike.builder()
                    .member(member)
                    .accommodation(accommodation)
                    .build();
            accomLikeRepository.save(newLike);
            return true; // 찜 등록
        }
    }

    public List<StayResponseDto> getMyWishlist(String email, LocalDate checkIn, LocalDate checkOut, Integer guests) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        Long stayDays = 0L;
        if (checkIn != null && checkOut != null && checkOut.isAfter(checkIn)) {
            stayDays = java.time.temporal.ChronoUnit.DAYS.between(checkIn, checkOut);
        } else {
            checkIn = LocalDate.now();
            checkOut = LocalDate.now();
        }

        return accomLikeRepository.findWishlistStaysByMemberIdAndDates(member.getMemberId(), checkIn, checkOut, stayDays, guests);
    }
}

import org.springframework.data.jpa.repository.Modifying;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

import java.time.LocalDate;
import java.util.List;

public interface RoomInventoryRepository extends JpaRepository<RoomInventory, Long> {

    // 특정 객실의 특정 날짜 구간 재고 조회
    @Query("SELECT ri FROM RoomInventory ri WHERE ri.room.roomId = :roomId AND ri.inventoryDate BETWEEN :checkIn AND :checkOut")
    List<RoomInventory> findInventoriesForReservation(
            @Param("roomId") Long roomId,
            @Param("checkIn") LocalDate checkIn,
            @Param("checkOut") LocalDate checkOut
    );
    // 💡 Fetch Join을 사용해 RoomInventory를 가져올 때 Room 정보도 한 번에(N+1 방지) 긁어옵니다.
    @Query("SELECT ri FROM RoomInventory ri JOIN FETCH ri.room r " +
            "WHERE r.accommodation.contentId = :contentId " + // ⚠️ Room 엔티티 안의 Accommodation 필드명에 맞게 수정!
            "AND ri.inventoryDate = :date " +
            "AND ri.availableCount > 0")
    List<RoomInventory> findAvailableRooms(
            @Param("contentId") Long contentId,
            @Param("date") LocalDate date
    );

    // 특정 숙소(String contentId)의 체크인 ~ 체크아웃 구간 객실별 재고 조회
    @Query("SELECT ri FROM RoomInventory ri " +
            "WHERE ri.room.accommodation.contentId = :contentId " +
            "  AND ri.inventoryDate >= :checkIn " +
            "  AND ri.inventoryDate < :checkOut")
    List<RoomInventory> findByContentIdAndDateRange(
            @Param("contentId") String contentId,
            @Param("checkIn") LocalDate checkIn,
            @Param("checkOut") LocalDate checkOut
    );

    // 💡JPQL 벌크 연산 도입: 특정 객실(Room)에 딸린 90일치 재고를 한 방에 삭제!
    @Modifying(clearAutomatically = true)
    @Query("DELETE FROM RoomInventory ri WHERE ri.room.roomId = :roomId")
    void deleteByRoom_RoomIdBulk(@Param("roomId") Long roomId);

    // 💡 특정 숙소(Accommodation)에 딸린 모든 객실의 모든 재고를 통째로 한 방에 삭제!
    @Modifying(clearAutomatically = true)
    @Query("DELETE FROM RoomInventory ri WHERE ri.room.accommodation.contentId = :contentId")
    void deleteAllInventoriesByAccommodationIdBulk(@Param("contentId") String contentId);
}

package org.example.hotelbooking.domain.room.entity;

import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import jakarta.persistence.*;
import lombok.*;

@Entity
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@Table(name = "ROOM")
public class Room {

    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "ROOM_ID")
    private Long roomId;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "CONTENT_ID", nullable = false)
    private Accommodation accommodation;

    @Column(name = "ROOM_TITLE", length = 100)
    private String roomTitle;

    @Column(name = "ROOM_SIZE", length = 50)
    private String roomSize;

    @Column(name = "ROOM_BASE_COUNT", length = 20)
    private String roomBaseCount;

    @Column(name = "ROOM_MAX_COUNT", length = 20)
    private String roomMaxCount;

    @Column(name = "PRICE_PER_NIGHT")
    private Integer pricePerNight;

    // 💡 Room.builder() 사용을 위한 빌더 생성자
    @Builder
    public Room(Accommodation accommodation, String roomTitle, String roomSize,
                String roomBaseCount, String roomMaxCount, Integer pricePerNight) {
        this.accommodation = accommodation;
        this.roomTitle = roomTitle;
        this.roomSize = roomSize;
        this.roomBaseCount = roomBaseCount;
        this.roomMaxCount = roomMaxCount;
        this.pricePerNight = pricePerNight;
    }

    // 💡 객실 정보 및 1박 가격 수정 메서드
    public void updateRoom(String roomTitle, String roomSize, String roomBaseCount,
                           String roomMaxCount, Integer pricePerNight) {
        this.roomTitle = roomTitle;
        this.roomSize = roomSize;
        this.roomBaseCount = roomBaseCount;
        this.roomMaxCount = roomMaxCount;
        this.pricePerNight = pricePerNight;
    }
}

package org.example.hotelbooking.domain.room.entity;

import jakarta.persistence.*;
import lombok.AccessLevel;
import lombok.Builder;
import lombok.Getter;
import lombok.NoArgsConstructor;

import java.time.LocalDate;

@Entity
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@Table(name = "ROOM_INVENTORY")
public class RoomInventory {

    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "INVENTORY_ID")
    private Long inventoryId;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "ROOM_ID", nullable = false)
    private Room room;

    @Column(name = "INVENTORY_DATE", nullable = false)
    private LocalDate inventoryDate; // 예: 2026-10-01

    @Column(name = "AVAILABLE_COUNT", nullable = false)
    private Integer availableCount; // 남은 객실 수

    @Builder
    public RoomInventory(Room room, LocalDate inventoryDate, Integer availableCount) {
        this.room = room;
        this.inventoryDate = inventoryDate;
        this.availableCount = availableCount;
    }

    // 💡 동시성 환경에서 호출될 재고 차감 로직
    public void decreaseCount(int count) {
        if (this.availableCount - count < 0) {
            throw new IllegalArgumentException("해당 날짜의 객실이 모두 예약되었습니다.");
        }
        this.availableCount -= count;
    }
    // 💡 예약 취소 시 재고 복구 로직
    public void increaseCount(int count) {
        this.availableCount += count;
    }
}

package org.example.hotelbooking.domain.room.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.accommodation.repository.AccommodationRepository;
import org.example.hotelbooking.domain.room.dto.AvailableRoomResponse;
import org.example.hotelbooking.domain.room.dto.RoomManageDto;
import org.example.hotelbooking.domain.room.entity.Room;
import org.example.hotelbooking.domain.room.entity.RoomImage;
import org.example.hotelbooking.domain.room.entity.RoomInventory;
import org.example.hotelbooking.domain.room.repository.RoomImageRepository;
import org.example.hotelbooking.domain.room.repository.RoomInventoryRepository;
import org.example.hotelbooking.domain.room.repository.RoomRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class RoomService {

    private final RoomInventoryRepository roomInventoryRepository;
    private final RoomRepository roomRepository;
    private final RoomImageRepository roomImageRepository;
    private final AccommodationRepository accommodationRepository;

    public List<AvailableRoomResponse> getAvailableRooms(Long contentId, LocalDate date) {
        return roomInventoryRepository.findAvailableRooms(contentId, date)
                .stream()
                .map(AvailableRoomResponse::from)
                .collect(Collectors.toList());
    }

    // 💡 1. 특정 숙소의 객실 목록 조회 (호스트 관리 화면용)
    public List<RoomManageDto.Response> getRoomsByAccommodation(String contentId) {
        List<Room> rooms = roomRepository.findByAccommodation_ContentId(contentId);
        return rooms.stream().map(room -> {
            List<String> urls = roomImageRepository
                    .findByRoom_RoomIdOrderByDisplayOrderAsc(room.getRoomId())
                    .stream()
                    .map(RoomImage::getImageUrl)
                    .toList();
            return RoomManageDto.Response.of(room, urls);
        }).toList();
    }

    // 💡 2. 호스트: 새 객실 추가 (ROOM + ROOM_IMAGE + 향후 90일 ROOM_INVENTORY 자동 생성)
    @Transactional
    public Long createRoom(String email, String contentId, RoomManageDto.SaveRequest request) {
        Accommodation accom = validateHostOwnership(email, contentId);

        Room room = Room.builder()
                .accommodation(accom)
                .roomTitle(request.getRoomTitle())
                .roomSize(request.getRoomSize())
                .roomBaseCount(request.getRoomBaseCount())
                .roomMaxCount(request.getRoomMaxCount())
                .pricePerNight(request.getPricePerNight())
                .build();

        Room savedRoom = roomRepository.save(room);

        // 객실 이미지 저장
        saveRoomImages(savedRoom, request.getImageUrls());

        // 오늘부터 90일 치 기본 객실 재고(ROOM_INVENTORY) 자동 세팅
        int stock = (request.getInitialStock() != null && request.getInitialStock() > 0)
                ? request.getInitialStock()
                : 5;
        LocalDate today = LocalDate.now();
        List<RoomInventory> inventories = new ArrayList<>();
        for (int i = 0; i < 90; i++) {
            inventories.add(RoomInventory.builder()
                    .room(savedRoom)
                    .inventoryDate(today.plusDays(i))
                    .availableCount(stock)
                    .build());
        }
        roomInventoryRepository.saveAll(inventories);

        return savedRoom.getRoomId();
    }

    // 💡 3. 호스트: 기존 객실 정보 및 가격 수정
    @Transactional
    public void updateRoom(String email, Long roomId, RoomManageDto.SaveRequest request) {
        Room room = roomRepository.findById(roomId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 객실입니다. id=" + roomId));

        validateHostOwnership(email, room.getAccommodation().getContentId());

        room.updateRoom(
                request.getRoomTitle(),
                request.getRoomSize(),
                request.getRoomBaseCount(),
                request.getRoomMaxCount(),
                request.getPricePerNight()
        );

        // 이미지 목록이 전달된 경우 기존 이미지 교체
        if (request.getImageUrls() != null) {
            // 💡 끝에 Bulk 를 붙여서 새 메서드를 호출하도록 변경!
            roomImageRepository.deleteByRoom_RoomIdBulk(roomId);
            saveRoomImages(room, request.getImageUrls());
        }
    }

    // 💡 4. 호스트: 객실 삭제
    @Transactional
    public void deleteRoom(String email, Long roomId) {
        Room room = roomRepository.findById(roomId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 객실입니다. id=" + roomId));

        validateHostOwnership(email, room.getAccommodation().getContentId());

        // 🚀 for문이나 낱개 쿼리 없이 Bulk Delete 메서드 호출
        roomImageRepository.deleteByRoom_RoomIdBulk(roomId);
        roomInventoryRepository.deleteByRoom_RoomIdBulk(roomId);

        roomRepository.delete(room);
    }

    private Accommodation validateHostOwnership(String email, String contentId) {
        Accommodation accom = accommodationRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 숙소입니다."));
        if (accom.getHost() == null || !accom.getHost().getEmail().equals(email)) {
            throw new IllegalArgumentException("본인 소유의 숙소 객실만 관리할 수 있습니다.");
        }
        return accom;
    }

    private void saveRoomImages(Room room, List<String> imageUrls) {
        if (imageUrls == null || imageUrls.isEmpty()) return;
        int order = 1;
        for (String url : imageUrls) {
            if (url != null && !url.isBlank()) {
                roomImageRepository.save(RoomImage.builder()
                        .room(room)
                        .imageUrl(url.trim())
                        .displayOrder(order++)
                        .build());
            }
        }
    }
}

package org.example.hotelbooking.domain.notification.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.accommodation.entity.AccommodationStatus;
import org.example.hotelbooking.domain.accommodation.repository.AccommodationRepository;
import org.example.hotelbooking.domain.member.entity.Member;
import org.example.hotelbooking.domain.member.repository.MemberRepository;
import org.example.hotelbooking.domain.notification.dto.NotificationDto;
import org.example.hotelbooking.domain.reservation.entity.Reservation;
import org.example.hotelbooking.domain.reservation.repository.ReservationRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.ArrayList;
import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class NotificationQueryService {

    private final MemberRepository memberRepository;
    private final AccommodationRepository accommodationRepository;
    private final ReservationRepository reservationRepository;

    public List<NotificationDto> getNotifications(String email) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        List<NotificationDto> list = new ArrayList<>();
        String roleName = member.getRole().getRoleName();

        // 1. 숙소 관리자 (ROLE_HOST) 알림
        if ("ROLE_HOST".equals(roleName)) {
            // [알림 1] 숙소 등록 반려 내역
            List<Accommodation> rejectedList = accommodationRepository
                    .findByHost_MemberIdAndApprovalStatus(member.getMemberId(), AccommodationStatus.REJECTED);

            for (Accommodation a : rejectedList) {
                String reason = a.getRejectReason() != null ? a.getRejectReason() : "사유가 입력되지 않았습니다.";
                list.add(NotificationDto.builder()
                        .id("REJECT_" + a.getContentId())
                        .title("숙소 신청이 반려되었습니다.")
                        .message("숙소: " + a.getTitle() + " / 반려 사유 : " + reason)
                        .type("ACCOM_REJECT")
                        .createdAt(a.getCreatedAt())
                        .build());
            }

            // [알림 2] 숙소 예약 신청 알림 (추후 예약 신청 연동용)
            List<Reservation> hostReservations = reservationRepository.findReservationsByHostId(member.getMemberId());
            for (Reservation r : hostReservations) {
                // 최근 예약 알림 (예시: COMPLETED/RESERVED 상태)
                list.add(NotificationDto.builder()
                        .id("RES_" + r.getReservationId())
                        .title("새로운 숙소 예약이 접수되었습니다.")
                        .message("예약자: " + r.getMember().getName() + " (" + r.getCheckInDate() + " ~ " + r.getCheckOutDate() + ")")
                        .type("RESERVATION_NEW")
                        .createdAt(r.getCreatedAt())
                        .build());
            }
        }

        // 2. 총괄 관리자 (ROLE_ADMIN) 알림
        if ("ROLE_ADMIN".equals(roleName)) {
            // [알림 3] 신규 등록되어 심사 대기 중인 숙소 알림
            List<Accommodation> pendingList = accommodationRepository
                    .findByApprovalStatusOrderByCreatedAtDesc(AccommodationStatus.PENDING);

            for (Accommodation a : pendingList) {
                list.add(NotificationDto.builder()
                        .id("PENDING_" + a.getContentId())
                        .title("신규 숙소 승인 요청이 있습니다.")
                        .message("숙소명: " + a.getTitle() + " (호스트: " + a.getHost().getName() + ")")
                        .type("ACCOM_PENDING")
                        .createdAt(a.getCreatedAt())
                        .build());
            }
        }

        // 최신 등록 순 정렬
        list.sort((o1, o2) -> {
            if (o1.getCreatedAt() == null || o2.getCreatedAt() == null) return 0;
            return o2.getCreatedAt().compareTo(o1.getCreatedAt());
        });

        return list;
    }
}

import org.springframework.web.bind.annotation.*;
import java.util.Map;

@Slf4j
@RestController
@RequestMapping("/api/v1/payments")
@RequiredArgsConstructor
public class PaymentController {

    private final TossPaymentService tossPaymentService;
    private final ReservationLockFacade reservationLockFacade;
    private final MemberRepository memberRepository;

    @GetMapping("/client-key")
    public ResponseEntity<Map<String, String>> getClientKey() {
        return ResponseEntity.ok(Map.of("clientKey", tossPaymentService.getTossClientKey()));
    }

    @PostMapping("/confirm")
    public ResponseEntity<String> confirmPayment(
            @RequestBody PaymentConfirmRequest request,
            @AuthenticationPrincipal String email
    ) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자 정보를 찾을 수 없습니다."));

        // 1. 토스페이먼츠 서버로 결제 승인(Confirm) 요청 (돈이 빠져나감)
        tossPaymentService.confirmPayment(request.getPaymentKey(), request.getOrderId(), request.getAmount());

        // 2. 결제가 승인되었으니, 우리 DB에 락을 걸고 객실 예약을 시도합니다.
        try {
            reservationLockFacade.reserveRoomWithLock(
                    member.getMemberId(),
                    request.getRoomId(),
                    request.getCheckInDate(),
                    request.getCheckOutDate(),
                    request.getAmount()
            );
            return ResponseEntity.ok("결제 및 예약이 성공적으로 완료되었습니다.");

        } catch (Exception e) {
            // 🚨 3. 치명적 예외: 돈은 나갔는데, 그 찰나의 순간에 방이 매진되어 DB 에러가 터졌다면?
            log.error("예약 DB 저장 실패! 유저의 돈을 즉시 환불합니다. 사유: {}", e.getMessage());

            // 보상 트랜잭션: 토스 서버로 즉시 결제 취소 API를 쏴서 전액 환불 조치
            tossPaymentService.cancelPayment(request.getPaymentKey(), "객실 매진으로 인한 자동 환불");

            return ResponseEntity.status(HttpStatus.CONFLICT)
                    .body("결제가 진행되었으나 찰나의 순간에 객실이 매진되었습니다. 안전하게 전액 환불되었습니다.");
        }
    }

package org.example.hotelbooking.domain.payment.service;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.*;
import org.springframework.stereotype.Service;
import org.springframework.web.client.HttpStatusCodeException;
import org.springframework.web.client.RestTemplate;
import java.nio.charset.StandardCharsets;
import java.util.Base64;
import java.util.HashMap;
import java.util.Map;

@Slf4j
@Service
@RequiredArgsConstructor
public class TossPaymentService {

    private final RestTemplate restTemplate = new RestTemplate();

    @Value("${toss.secret.key}")
    private String tossSecretKey;

    @Value("${toss.client.key:}")
    private String tossClientKey;

    public String getTossClientKey() {
        return tossClientKey;
    }

    // 1. 결제 최종 승인 API 호출
    public void confirmPayment(String paymentKey, String orderId, Integer amount) {
        String url = "https://api.tosspayments.com/v1/payments/confirm";

        Map<String, Object> params = new HashMap<>();
        params.put("paymentKey", paymentKey);
        params.put("orderId", orderId);
        params.put("amount", amount);

        HttpEntity<Map<String, Object>> request = new HttpEntity<>(params, getHeaders());

        try {
            restTemplate.postForEntity(url, request, String.class);
            log.info("✅ 토스 결제 승인 완료 - OrderId: {}", orderId);

        } catch (HttpStatusCodeException e) {
            // 💡 추가된 부분: 토스 서버가 거절한 "진짜 이유(JSON)"를 콘솔에 출력합니다!
            log.error("❌ 토스 결제 승인 거절됨: {}", e.getResponseBodyAsString());
            throw new RuntimeException("결제 승인 거절: " + e.getResponseBodyAsString());

        } catch (Exception e) {
            log.error("❌ 토스 결제 승인 실패 (통신 오류): {}", e.getMessage());
            throw new RuntimeException("결제 승인 중 통신 오류가 발생했습니다.");
        }
    }

    // 2. 결제 취소(환불) API 호출 (보상 트랜잭션용)
    public void cancelPayment(String paymentKey, String cancelReason) {
        String url = "https://api.tosspayments.com/v1/payments/" + paymentKey + "/cancel";

        Map<String, Object> params = new HashMap<>();
        params.put("cancelReason", cancelReason);

        HttpEntity<Map<String, Object>> request = new HttpEntity<>(params, getHeaders());

        try {
            restTemplate.postForEntity(url, request, String.class);
            log.info("🔄 토스 결제 자동 취소(환불) 완료 - PaymentKey: {}, 사유: {}", paymentKey, cancelReason);
        } catch (Exception e) {
            log.error("❌ 토스 결제 자동 취소 실패 - 수동 환불 필요: {}", e.getMessage());
            // 실무에서는 여기에 슬랙 알림 등을 붙여서 관리자가 수동으로 환불하게 만듭니다.
        }
    }

    // 토스페이먼츠 Basic Auth 헤더 생성 (시크릿키 뒤에 콜론(:)을 붙여서 Base64 인코딩)
    private HttpHeaders getHeaders() {
        HttpHeaders headers = new HttpHeaders();
        String encodedAuth = Base64.getEncoder().encodeToString((tossSecretKey + ":").getBytes(StandardCharsets.UTF_8));
        headers.setBasicAuth(encodedAuth);
        headers.setContentType(MediaType.APPLICATION_JSON);
        return headers;
    }
}

package org.example.hotelbooking.domain.member.entity;

import jakarta.persistence.*;
import lombok.*;
import java.time.LocalDateTime;

@Entity
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@Table(name = "MEMBER")
public class Member {

    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "MEMBER_ID")
    private Long memberId;

    @Column(unique = true, nullable = false, length = 100)
    private String email;

    @Column(nullable = false, length = 255)
    private String password;

    @Column(nullable = false, length = 50)
    private String name;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "ROLE_ID")
    private Role role;

    @Column(name = "BUSINESS_NO", unique = true, length = 50)
    private String businessNo;

    // 💡 DB의 STATUS 컬럼 매핑 ('ACTIVE', 'BLOCKED', 'WITHDRAWN', 'BANNED')
    @Column(name = "STATUS", length = 20, nullable = false)
    private String status;

    @Column(name = "CREATED_AT", updatable = false)
    private LocalDateTime createdAt;

    @PrePersist
    protected void onCreate() {
        this.createdAt = LocalDateTime.now();
        if (this.status == null) {
            this.status = "ACTIVE";
        }
    }

    @Builder
    public Member(String email, String password, String name, Role role, String businessNo, String status) {
        this.email = email;
        this.password = password;
        this.name = name;
        this.role = role;
        this.businessNo = businessNo;
        this.status = (status != null) ? status : "ACTIVE";
    }

    public void updateToHost(String businessNo, Role role) {
        this.businessNo = businessNo;
        this.role = role;
    }
    public void updatePassword(String newPassword){
        this.password = newPassword;
    }

    public void updateInfo(String name) {
        this.name = name;
    }

    public void downgradeToUser(Role userRole) {
        this.role = userRole;
        this.businessNo = null; // 사업자번호 초기화
    }

    // 💡 회원 탈퇴 처리 편의 메서드
    // 1) status를 'WITHDRAWN'으로 변경
    // 2) 동일 이메일 재가입이 가능하도록 email을 고유한 unknown 형태로 변경 (UNIQUE 충돌 방지)
    // 3) 이름을 'Unknown'으로 변경 및 사업자번호 null 초기화
    public void withdraw(Role userRole) {
        this.status = "WITHDRAWN";
        this.name = "Unknown";
        this.email = "unknown_" + this.memberId + "_" + System.currentTimeMillis() + "@withdrawn.com";
        this.businessNo = null;
        if (userRole != null) {
            this.role = userRole;
        }
    }
}

package org.example.hotelbooking.domain.member.controller;

import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.member.dto.AuthDto;
import org.example.hotelbooking.domain.member.service.AuthService;
import org.example.hotelbooking.domain.member.service.KakaoAuthService;
import org.springframework.beans.factory.annotation.Value; // 💡 이거 꼭 필요합니다!
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseCookie;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.web.bind.annotation.*;
import java.util.Map;
import java.net.URI;

@RestController
@RequestMapping("/api/v1/auth")
@RequiredArgsConstructor
public class AuthController {

    private final AuthService authService;
    private final KakaoAuthService kakaoAuthService;

    // 💡 application.properties의 app.domain 값을 가져오되, 없으면 5173 사용
    @Value("${app.domain:http://localhost:5173}")
    private String appDomain;

    @PostMapping("/signup")
    public ResponseEntity<String> signup(@RequestBody AuthDto.SignupRequest request) {
        authService.signup(request);
        return ResponseEntity.ok("회원가입이 완료되었습니다.");
    }

    @PostMapping("/login")
    public ResponseEntity<AuthDto.TokenResponse> login(@RequestBody AuthDto.LoginRequest request, HttpServletResponse response) {
        AuthDto.TokenResponse tokenResponse = authService.login(request);

        // 💡 Refresh Token을 HttpOnly 쿠키로 생성
        ResponseCookie cookie = ResponseCookie.from("refreshToken", tokenResponse.getRefreshToken())
                .httpOnly(true)
                .secure(false) // 🚨 나중에 HTTPS(도메인) 적용 시 반드시 true로 변경!
                .path("/")
                .maxAge(7 * 24 * 60 * 60) // 7일
                .sameSite("Lax")
                .build();

        // 헤더에 쿠키 추가
        response.addHeader(HttpHeaders.SET_COOKIE, cookie.toString());

        return ResponseEntity.ok(tokenResponse);
    }

    // 💡 수정한 부분: 하드코딩 대신 appDomain을 사용하여 동적으로 리다이렉트!
    @GetMapping("/kakao/callback")
    public ResponseEntity<Void> kakaoCallback(@RequestParam String code) {
        AuthDto.TokenResponse tokens = kakaoAuthService.kakaoLogin(code);

        // 1. Refresh Token을 HttpOnly 쿠키로 생성
        ResponseCookie cookie = ResponseCookie.from("refreshToken", tokens.getRefreshToken())
                .httpOnly(true)
                .secure(false) // 현재 HTTP 상태이므로 false
                .path("/")
                .maxAge(7 * 24 * 60 * 60) // 7일
                .sameSite("Lax")
                .build();

        HttpHeaders headers = new HttpHeaders();
        // 2. 응답 헤더에 쿠키 세팅
        headers.add(HttpHeaders.SET_COOKIE, cookie.toString());
        // 3. 리다이렉트 URL에는 accessToken만 깔끔하게 넘김 (refreshToken 제거)
        headers.setLocation(URI.create(appDomain + "/oauth/callback?token=" + tokens.getToken()));

        return new ResponseEntity<>(headers, HttpStatus.FOUND);
    }

    // 💡 Refresh Token은 이제 Request Body가 아닌 쿠키에서 읽어옵니다.
    @PostMapping("/refresh")
    public ResponseEntity<AuthDto.TokenResponse> refresh(
            @CookieValue(value = "refreshToken", required = false) String refreshToken) {

        if (refreshToken == null || refreshToken.isBlank()) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED).build();
        }

        AuthDto.TokenResponse newTokens = authService.reissueToken(refreshToken);
        return ResponseEntity.ok(newTokens);
    }

    @PostMapping("/apply-host")
    public ResponseEntity<AuthDto.TokenResponse> applyHost(
            @RequestBody AuthDto.HostApplyRequest request,
            @AuthenticationPrincipal String email
    ) {
        if (email == null) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED).build();
        }
        // 💡 AuthService가 이미 2개가 담긴 TokenResponse를 주니까 그대로 받아서 넘깁니다!
        AuthDto.TokenResponse tokens = authService.upgradeToHost(email, request.getBusinessNo());
        return ResponseEntity.ok(tokens);
    }
}

@Slf4j
@Service
@RequiredArgsConstructor
public class MailService {

    private final JavaMailSender javaMailSender;
    private final TemplateEngine templateEngine; // 💡 Thymeleaf 템플릿 엔진 주입
    @Value("${app.domain:http://localhost:5173}")
    private String appDomain;

    @Value("${spring.mail.username}")
    private String fromEmail;

    public void sendAuthCode(String toEmail, String code) {
        MimeMessage mimeMessage = javaMailSender.createMimeMessage();

        try {
            MimeMessageHelper helper = new MimeMessageHelper(mimeMessage, false, "UTF-8");

            helper.setFrom(fromEmail, "제주 야호");
            helper.setTo(toEmail);
            helper.setSubject("[제주 야호] 비밀번호 찾기 인증번호 안내");

            // 💡 템플릿 변수 매핑 및 HTML 렌더링
            Context context = new Context();
            context.setVariable("code", code);
            String htmlContent = templateEngine.process("mail/password-reset", context);

            helper.setText(htmlContent, true);

            javaMailSender.send(mimeMessage);
            log.info("인증 메일 발송 완료 -> 수신자: {}", toEmail);

        } catch (MessagingException | UnsupportedEncodingException e) {
            log.error("메일 발송 중 오류 발생: {}", e.getMessage(), e);
            throw new RuntimeException("이메일 발송에 실패했습니다. 관리자에게 문의하세요.", e);
        }
    }

    // 💡 2. [신규 추가] 숙소 예약 완료 안내 이메일 발송 (비동기 처리)
    @Async
    public void sendReservationConfirmation(
            String toEmail,
            String guestName,
            Long reservationId,
            String contentId,
            String stayTitle,
            String stayAddr,
            String stayTel,
            String checkinTime,
            String checkoutTime,
            String roomTitle,
            LocalDate checkInDate,
            LocalDate checkOutDate,
            Integer totalPrice
    ) {
        // 카카오 가짜 이메일(@hotel.com)인 경우 실제 발송 생략
        if (toEmail == null || (toEmail.startsWith("kakao_") && toEmail.endsWith("@hotel.com"))) {
            log.info("소셜 로그인 가짜 이메일 계정이므로 예약 확인 메일 발송을 생략합니다: {}", toEmail);
            return;
        }

        MimeMessage mimeMessage = javaMailSender.createMimeMessage();

        try {
            MimeMessageHelper helper = new MimeMessageHelper(mimeMessage, false, "UTF-8");

            helper.setFrom(fromEmail, "제주 야호");
            helper.setTo(toEmail);
            helper.setSubject("[제주 야호] '" + stayTitle + "' 숙소 예약이 확정되었습니다.");

            long nights = Math.max(1, ChronoUnit.DAYS.between(checkInDate, checkOutDate));
            String reservationNo = "#" + checkInDate.toString().replace("-", "") + "-" + String.format("%04d", reservationId);
            String formattedPrice = NumberFormat.getNumberInstance(Locale.KOREA).format(totalPrice != null ? totalPrice : 0) + "원";

            Context context = new Context();
            context.setVariable("guestName", guestName);
            context.setVariable("reservationNo", reservationNo);
            context.setVariable("stayTitle", stayTitle);
            context.setVariable("stayAddr", stayAddr != null ? stayAddr : "제주특별자치도");
            context.setVariable("stayTel", stayTel != null ? stayTel : "숙소 문의");
            context.setVariable("checkinTime", checkinTime != null ? checkinTime : "15:00");
            context.setVariable("checkoutTime", checkoutTime != null ? checkoutTime : "11:00");
            context.setVariable("roomTitle", roomTitle);
            context.setVariable("checkInDate", checkInDate.toString());
            context.setVariable("checkOutDate", checkOutDate.toString());
            context.setVariable("nights", nights);
            context.setVariable("totalPriceFormatted", formattedPrice);

            // 💡 Localhost 프론트엔드 바로가기 링크 세팅
            context.setVariable("myReservationsUrl", appDomain + "/my-reservations");
            context.setVariable("stayDetailUrl", appDomain + "/stays/" + contentId);

            String htmlContent = templateEngine.process("mail/reservation-confirm", context);
            helper.setText(htmlContent, true);

            javaMailSender.send(mimeMessage);
            log.info("✅ 예약 확정 이메일 발송 완료 -> 수신자: {}, 예약번호: {}", toEmail, reservationNo);

        } catch (Exception e) {
            // 이메일 발송 실패가 예약 트랜잭션을 롤백시키지 않도록 로그만 기록
            log.error("❌ 예약 확정 이메일 발송 실패 (수신자: {}): {}", toEmail, e.getMessage());
        }
    }
}

    public void sendPasswordResetCode(String email) {
        // 💡 [신규 추가] 1. 이메일 발송 1분 쿨타임(속도 제한) 검증
        String limitKey = "PWD_RESET_LIMIT:" + email;
        if (Boolean.TRUE.equals(redisTemplate.hasKey(limitKey))) {
            throw new IllegalArgumentException("요청이 너무 많습니다. 1분 후에 다시 시도해 주세요.");
        }

        // 2. 기존 로직: 이메일로 회원 정보 조회
        org.example.hotelbooking.domain.member.entity.Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("가입되지 않은 이메일입니다."));

        // 3. 기존 로직: 소셜 가입자 판별
        if (email.startsWith("kakao_") && email.endsWith("@hotel.com")) {
            throw new IllegalArgumentException("카카오 소셜 로그인으로 가입된 계정입니다. 카카오 로그인을 이용해 주세요.");
        }

        // 4. 인증 코드 생성 및 Redis 저장 (기존 3분)
        String code = String.valueOf((int) (Math.random() * 899999) + 100000);
        String redisKey = "PWD_RESET:" + email;
        redisTemplate.opsForValue().set(redisKey, code, 3, java.util.concurrent.TimeUnit.MINUTES);

        // 💡 [신규 추가] 5. 성공적으로 처리되었으므로 1분(60초) 동안 재요청을 막는 Lock 설정
        redisTemplate.opsForValue().set(limitKey, "LOCKED", 1, java.util.concurrent.TimeUnit.MINUTES);

        // 6. 이메일 발송
        mailService.sendAuthCode(email, code);
    }
    // 💡 2번 API: 인증 코드만 검증하고 합격 티켓 부여
    public void verifyResetCode(String email, String inputCode) {
        String redisKey = "PWD_RESET:" + email;
        String savedCode = redisTemplate.opsForValue().get(redisKey);

        if (savedCode == null || !savedCode.equals(inputCode)) {
            throw new IllegalArgumentException("인증 코드가 일치하지 않거나 만료되었습니다.");
        }

        // 인증 성공 시, 기존 인증 코드는 삭제
        redisTemplate.delete(redisKey);

        // "이메일 인증 완료" 상태를 Redis에 10분간 저장 (합격 티켓)
        redisTemplate.opsForValue().set("PWD_RESET_VERIFIED:" + email, "TRUE", 10, java.util.concurrent.TimeUnit.MINUTES);
    }

    // 💡 3번 API: 합격 티켓 확인 후 실제 비밀번호 변경
    @org.springframework.transaction.annotation.Transactional
    public void resetPassword(String email, String newPassword) {
        String verifiedKey = "PWD_RESET_VERIFIED:" + email;
        String isVerified = redisTemplate.opsForValue().get(verifiedKey);

        // 1. 합격 티켓이 없으면 에러 (인증 안 하고 바로 변경 시도한 경우)
        if (isVerified == null) {
            throw new IllegalArgumentException("이메일 인증이 완료되지 않았거나 만료되었습니다. 다시 인증해 주세요.");
        }

        // 2. 회원 조회 및 비밀번호 변경
        org.example.hotelbooking.domain.member.entity.Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("가입되지 않은 이메일입니다."));

        // 💡 실무 주의: Spring Security를 쓰고 계시다면 passwordEncoder.encode(newPassword)로 암호화해야 합니다!
        member.updatePassword(passwordEncoder.encode(newPassword));

        // 3. 비밀번호 변경이 끝났으므로 합격 티켓 삭제 (재사용 방지)
        redisTemplate.delete(verifiedKey);
    }

    @Transactional(readOnly = true)

    // 💡 2. 회원정보 수정 (이름, 비밀번호)
    @Transactional
    public AuthDto.TokenResponse updateProfile(String email, MemberDto.UpdateRequest request) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        // 이름 변경
        if (request.getName() != null && !request.getName().isBlank()) {
            member.updateInfo(request.getName().trim());
        }

        // 비밀번호 변경 (입력된 경우에만 검증 후 변경)
        if (request.getNewPassword() != null && !request.getNewPassword().isBlank()) {
            boolean isSocial = email.startsWith("kakao_") && email.endsWith("@hotel.com");
            if (isSocial) {
                throw new IllegalArgumentException("소셜 로그인 회원은 비밀번호를 변경할 수 없습니다.");
            }

            if (request.getCurrentPassword() == null || !passwordEncoder.matches(request.getCurrentPassword(), member.getPassword())) {
                throw new IllegalArgumentException("현재 비밀번호가 일치하지 않습니다.");
            }

            member.updatePassword(passwordEncoder.encode(request.getNewPassword()));
        }

        // 변경된 닉네임이 반영된 새 JWT 토큰 발급
        // 💡 변경된 닉네임이 반영된 새 JWT 액세스/리프레시 토큰 발급 및 Redis 저장
        String newAccessToken = jwtTokenProvider.createToken(member.getEmail(), member.getRole().getRoleName(), member.getName());
        String newRefreshToken = jwtTokenProvider.createRefreshToken(member.getEmail());

        redisTemplate.opsForValue().set("RT:" + member.getEmail(), newRefreshToken, 7, TimeUnit.DAYS);

        return new AuthDto.TokenResponse(newAccessToken, newRefreshToken);
    }

    // 💡 3. 숙소 관리자 탈퇴 (보유 숙소 또는 진행 예정 예약이 있으면 차단!)
    @Transactional
    public AuthDto.TokenResponse downgradeHost(String email) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        if (!"ROLE_HOST".equals(member.getRole().getRoleName())) {
            throw new IllegalArgumentException("숙소 관리자 권한을 가진 회원만 신청할 수 있습니다.");
        }

        // [검증 1] 내 숙소에 투숙 예정/진행 중인 예약이 남아있는지 확인
        LocalDate today = LocalDate.now();
        List<Reservation> hostReservations = reservationRepository.findReservationsByHostId(member.getMemberId());
        boolean hasUpcomingReservation = hostReservations.stream()
                .anyMatch(r -> !"CANCELED".equals(r.getStatus()) && r.getCheckOutDate().isAfter(today));

        if (hasUpcomingReservation) {
            throw new IllegalStateException("현재 투숙 예정이거나 진행 중인 예약이 존재하여 숙소 관리자 자격을 해지할 수 없습니다.");
        }

        // [검증 2] 등록된 숙소가 존재하는지 확인
        List<Accommodation> myAccommodations = accommodationRepository.findByHost_Email(email);
        if (!myAccommodations.isEmpty()) {
            throw new IllegalStateException(
                    "등록된 숙소(" + myAccommodations.size() + "곳)가 존재하여 숙소 관리자 자격을 해지할 수 없습니다. 더이상 서비스하지 않는 경우 숙소 삭제 후 해지 바랍니다."
            );
        }

        Role userRole = roleRepository.findByRoleName("ROLE_USER")
                .orElseThrow(() -> new IllegalArgumentException("ROLE_USER 권한이 존재하지 않습니다."));

        member.downgradeToUser(userRole);

        String newAccessToken = jwtTokenProvider.createToken(member.getEmail(), userRole.getRoleName(), member.getName());
        String newRefreshToken = jwtTokenProvider.createRefreshToken(member.getEmail());

        redisTemplate.opsForValue().set("RT:" + member.getEmail(), newRefreshToken, 7, TimeUnit.DAYS);

        return new AuthDto.TokenResponse(newAccessToken, newRefreshToken);
    }

    // 💡 4. 회원 탈퇴 (status -> WITHDRAWN, 이름/이메일 Unknown 처리)
    @Transactional
    public void withdrawMember(String email) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        LocalDate today = LocalDate.now();

        // [검증 1] 일반 유저(게스트)로서 체크아웃 전인 진행 예정 예약이 있는지 확인
        List<Reservation> myGuestReservations = reservationRepository
                .findByMember_MemberIdOrderByCreatedAtDesc(member.getMemberId());
        boolean hasUpcomingGuestRes = myGuestReservations.stream()
                .anyMatch(r -> !"CANCELED".equals(r.getStatus()) && r.getCheckOutDate().isAfter(today));

        if (hasUpcomingGuestRes) {
            throw new IllegalStateException("이용 예정이거나 투숙 중인 예약이 존재하여 탈퇴할 수 없습니다. 예약을 먼저 취소해 주세요.");
        }

        // [검증 2] 호스트(ROLE_HOST)인 경우 보유 숙소나 호스트 예약이 남아있는지 확인
        if ("ROLE_HOST".equals(member.getRole().getRoleName())) {
            List<Reservation> hostReservations = reservationRepository.findReservationsByHostId(member.getMemberId());
            boolean hasUpcomingHostRes = hostReservations.stream()
                    .anyMatch(r -> !"CANCELED".equals(r.getStatus()) && r.getCheckOutDate().isAfter(today));

            List<Accommodation> myAccommodations = accommodationRepository.findByHost_Email(email);

            if (hasUpcomingHostRes || !myAccommodations.isEmpty()) {
                throw new IllegalStateException("보유 중인 숙소 또는 진행 예정인 예약이 존재하여 탈퇴할 수 없습니다. 숙소 삭제 후 다시 시도해 주세요.");
            }
        }

        Role userRole = roleRepository.findByRoleName("ROLE_USER").orElse(member.getRole());

        // 상태 WITHDRAWN 변경 및 이메일/이름 Unknown 처리 (기존 예약·리뷰 FK는 깨지지 않고 유지됨!)
        member.withdraw(userRole);
    }
}

import java.util.concurrent.TimeUnit;

@Service
@RequiredArgsConstructor
public class AuthService {

    private final MemberRepository memberRepository;
    private final RoleRepository roleRepository;
    private final PasswordEncoder passwordEncoder;
    private final JwtTokenProvider jwtTokenProvider;
    private final StringRedisTemplate redisTemplate;

    @Transactional
    public void signup(AuthDto.SignupRequest request) {
        if (memberRepository.existsByEmail(request.getEmail())) {
            throw new IllegalArgumentException("이미 가입된 이메일입니다.");
        }

        // 💡 1. 사업자등록번호 유무에 따라 부여할 권한 결정 (호스트 체크 시 ROLE_HOST, 일반 시 ROLE_USER)
        boolean isHost = request.getBusinessNo() != null && !request.getBusinessNo().trim().isEmpty();
        String targetRoleName = isHost ? "ROLE_HOST" : "ROLE_USER";

        Role assignedRole = roleRepository.findByRoleName(targetRoleName)
                .orElseThrow(() -> new IllegalArgumentException("시스템 에러: " + targetRoleName + " 권한이 존재하지 않습니다."));

        // 💡 2. 동적으로 결정된 assignedRole 세팅 및 Member 엔티티 조립
        Member member = Member.builder()
                .email(request.getEmail())
                .password(passwordEncoder.encode(request.getPassword())) // BCrypt 암호화!
                .name(request.getName())
                .businessNo(isHost ? request.getBusinessNo().trim() : null)
                .role(assignedRole) // ✅ 호스트 가입 시 ROLE_HOST (ROLE_ID: 2) 할당
                .status("ACTIVE")
                .build();

        // 3. DB에 저장
        memberRepository.save(member);
    }

    // 💡 로그인 시 Access + Refresh Token 모두 발급 및 Redis 저장
    @Transactional(readOnly = true)
    public AuthDto.TokenResponse login(AuthDto.LoginRequest request) {
        // 💡 1. 로그인 실패 횟수 조회
        String failKey = "LOGIN_FAIL:" + request.getEmail();
        String failCountStr = redisTemplate.opsForValue().get(failKey);
        int failCount = failCountStr != null ? Integer.parseInt(failCountStr) : 0;

        // 💡 2. 5회 이상 실패 시 즉시 차단
        if (failCount >= 5) {
            throw new IllegalArgumentException("비밀번호 5회 오류로 30분간 로그인이 제한됩니다.");
        }

        Member member = memberRepository.findByEmail(request.getEmail())
                .orElseThrow(() -> new IllegalArgumentException("가입되지 않은 이메일입니다."));

        if ("WITHDRAWN".equals(member.getStatus())) {
            throw new IllegalArgumentException("이미 탈퇴 처리된 계정입니다.");
        }

        // 💡 3. 비밀번호 불일치 시 카운트 증가
        if (!passwordEncoder.matches(request.getPassword(), member.getPassword())) {
            failCount++;
            redisTemplate.opsForValue().set(failKey, String.valueOf(failCount), 30, TimeUnit.MINUTES);

            if (failCount >= 5) {
                throw new IllegalArgumentException("비밀번호 5회 오류로 30분간 로그인이 제한됩니다.");
            } else {
                throw new IllegalArgumentException("비밀번호가 일치하지 않습니다. (" + failCount + "/5)");
            }
        }

        // 💡 4. 로그인 성공 시 실패 카운트 초기화
        redisTemplate.delete(failKey);

        // 5. 토큰 2개 발급
        String accessToken = jwtTokenProvider.createToken(member.getEmail(), member.getRole().getRoleName(), member.getName());
        String refreshToken = jwtTokenProvider.createRefreshToken(member.getEmail());

        // Redis에 Refresh Token 저장
        redisTemplate.opsForValue().set("RT:" + member.getEmail(), refreshToken, 7, TimeUnit.DAYS);

        return new AuthDto.TokenResponse(accessToken, refreshToken);
    }

    @Transactional(readOnly = true)
    public AuthDto.TokenResponse reissueToken(String refreshToken) {
        // 1. 넘어온 RT 유효성 검증
        if (!jwtTokenProvider.validateToken(refreshToken)) {
            throw new IllegalArgumentException("Refresh Token이 만료되었거나 유효하지 않습니다.");
        }

        // 2. 토큰에서 이메일 추출
        String email = jwtTokenProvider.getUserEmail(refreshToken);

        // 3. Redis에 저장된 진짜 RT와 비교
        String savedToken = redisTemplate.opsForValue().get("RT:" + email);
        if (savedToken == null || !savedToken.equals(refreshToken)) {
            throw new IllegalArgumentException("변조되거나 로그아웃된 Refresh Token입니다.");
        }

        // 4. 문제없다면 새 Access Token 생성 후 반환 (RT는 기존 것 그대로 재사용)
        Member member = memberRepository.findByEmail(email).orElseThrow();
        String newAccessToken = jwtTokenProvider.createToken(email, member.getRole().getRoleName(), member.getName());

        return new AuthDto.TokenResponse(newAccessToken, refreshToken);
    }
    // 유저에서 숙소관리자 신청 후 숙소관리자(ROLE_HOST) 권한 부여 로직
    @Transactional
    public AuthDto.TokenResponse upgradeToHost(String email, String businessNo) {
        Member member = memberRepository.findByEmail(email).orElseThrow();
        Role hostRole = roleRepository.findByRoleName("ROLE_HOST").orElseThrow();
        member.updateToHost(businessNo, hostRole);

        // 호스트 전환 시에도 두 토큰 모두 갱신
        String accessToken = jwtTokenProvider.createToken(member.getEmail(), hostRole.getRoleName(), member.getName());
        String refreshToken = jwtTokenProvider.createRefreshToken(member.getEmail());
        redisTemplate.opsForValue().set("RT:" + member.getEmail(), refreshToken, 7, TimeUnit.DAYS);

        return new AuthDto.TokenResponse(accessToken, refreshToken);
    }
}

package org.example.hotelbooking.domain.member.service;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.member.dto.AuthDto;
import org.example.hotelbooking.domain.member.entity.Member;

import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpMethod;
import org.springframework.http.ResponseEntity;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.util.LinkedMultiValueMap;
import org.springframework.util.MultiValueMap;
import org.springframework.web.client.RestTemplate;

import java.util.UUID;
import java.util.concurrent.TimeUnit;

@Service
@RequiredArgsConstructor
public class KakaoAuthService {

    @Value("${kakao.client.id}")
    private String clientId;

    @Value("${kakao.redirect.uri}")
    private String redirectUri;

    private final MemberRepository memberRepository;
    private final RoleRepository roleRepository;
    private final PasswordEncoder passwordEncoder;
    private final JwtTokenProvider jwtTokenProvider;
    private final StringRedisTemplate redisTemplate;

    @Transactional
    public AuthDto.TokenResponse kakaoLogin(String code) {
        // 1. 카카오 서버로 인가 코드를 보내서 '액세스 토큰'을 받아옵니다.
        String accessToken = getAccessToken(code);

        // 2. 받아온 액세스 토큰으로 카카오 유저 정보(고유번호, 닉네임)를 가져옵니다.
        JsonNode userInfo = getUserInfo(accessToken);
        String kakaoId = userInfo.get("id").asText();
        String nickname = userInfo.get("properties").get("nickname").asText();

        // 3. 이메일을 못 받으므로, 카카오 고유번호를 이용해 가짜 이메일을 생성합니다.
        String dummyEmail = "kakao_" + kakaoId + "@hotel.com";

        // 4. DB에 이미 있는 유저인지 확인하고, 없으면 자동 회원가입 시킵니다.
        Member member = memberRepository.findByEmail(dummyEmail).orElseGet(() -> {
            Role userRole = roleRepository.findByRoleName("ROLE_USER")
                    .orElseThrow(() -> new IllegalArgumentException("기본 권한이 없습니다."));

            // 비밀번호는 랜덤 UUID로 생성해서 암호화 (어차피 카카오 로그인은 비밀번호를 안 씁니다)
            Member newMember = Member.builder()
                    .email(dummyEmail)
                    .password(passwordEncoder.encode(UUID.randomUUID().toString()))
                    .name(nickname)
                    .role(userRole)
                    .status("ACTIVE")
                    .build();
            return memberRepository.save(newMember);
        });

        // 토큰 2개 생성 및 Redis 저장
        String newAccessToken = jwtTokenProvider.createToken(member.getEmail(), member.getRole().getRoleName(), member.getName());
        String newRefreshToken = jwtTokenProvider.createRefreshToken(member.getEmail());
        redisTemplate.opsForValue().set("RT:" + member.getEmail(), newRefreshToken, 7, TimeUnit.DAYS);

        return new AuthDto.TokenResponse(newAccessToken, newRefreshToken);
    }

    // --- 카카오 API 통신 메서드 ---

    private String getAccessToken(String code) {
        RestTemplate rt = new RestTemplate();
        HttpHeaders headers = new HttpHeaders();
        headers.add("Content-type", "application/x-www-form-urlencoded;charset=utf-8");

        MultiValueMap<String, String> params = new LinkedMultiValueMap<>();
        params.add("grant_type", "authorization_code");
        params.add("client_id", clientId);
        params.add("redirect_uri", redirectUri);
        params.add("code", code);

        HttpEntity<MultiValueMap<String, String>> request = new HttpEntity<>(params, headers);
        ResponseEntity<String> response = rt.postForEntity("https://kauth.kakao.com/oauth/token", request, String.class);

        try {
            ObjectMapper mapper = new ObjectMapper();
            JsonNode node = mapper.readTree(response.getBody());
            return node.get("access_token").asText();
        } catch (Exception e) {
            throw new RuntimeException("카카오 토큰 발급 중 오류가 발생했습니다.", e);
        }
    }

    private JsonNode getUserInfo(String accessToken) {
        RestTemplate rt = new RestTemplate();
        HttpHeaders headers = new HttpHeaders();
        headers.add("Authorization", "Bearer " + accessToken);
        headers.add("Content-type", "application/x-www-form-urlencoded;charset=utf-8");

        HttpEntity<MultiValueMap<String, String>> request = new HttpEntity<>(headers);
        ResponseEntity<String> response = rt.exchange("https://kapi.kakao.com/v2/user/me", HttpMethod.GET, request, String.class);

        try {
            ObjectMapper mapper = new ObjectMapper();
            return mapper.readTree(response.getBody());
        } catch (Exception e) {
            throw new RuntimeException("카카오 유저 정보 조회 중 오류가 발생했습니다.", e);
        }
    }
}

package org.example.hotelbooking.domain.reservation.entity;

import org.example.hotelbooking.domain.member.entity.Member;
import org.example.hotelbooking.domain.room.entity.Room;
import jakarta.persistence.*;
import lombok.*;
import java.time.LocalDate;
import java.time.LocalDateTime;

@Entity
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@Table(name = "RESERVATION")
public class Reservation {

    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "RESERVATION_ID")
    private Long reservationId;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "MEMBER_ID", nullable = false)
    private Member member;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "ROOM_ID", nullable = false)
    private Room room;

    @Column(name = "CHECK_IN_DATE", nullable = false)
    private LocalDate checkInDate;

    @Column(name = "CHECK_OUT_DATE", nullable = false)
    private LocalDate checkOutDate;

    @Column(name = "TOTAL_PRICE", nullable = false)
    private Integer totalPrice;

    @Column(length = 20)
    private String status;

    @Column(name = "CREATED_AT", updatable = false)
    private LocalDateTime createdAt;

    @PrePersist
    protected void onCreate() {
        this.createdAt = LocalDateTime.now();
        if (this.status == null) this.status = "COMPLETED";
    }
    // 빌더 추가
    @Builder
    public Reservation(Member member, Room room, LocalDate checkInDate, LocalDate checkOutDate, Integer totalPrice, String status) {
        this.member = member;
        this.room = room;
        this.checkInDate = checkInDate;
        this.checkOutDate = checkOutDate;
        this.totalPrice = totalPrice;
        this.status = status;
    }
    // 💡 예약 취소 로직
    public void cancel() {
        if ("CANCELED".equals(this.status)) {
            throw new IllegalStateException("이미 취소된 예약입니다.");
        }
        this.status = "CANCELED";
    }
}

package org.example.hotelbooking.domain.reservation.facade;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.example.hotelbooking.domain.reservation.repository.ReservationRepository;
import org.example.hotelbooking.domain.reservation.service.ReservationService;
import org.redisson.api.RLock;
import org.redisson.api.RedissonClient;
import org.springframework.stereotype.Component;

import java.time.LocalDate;
import java.util.concurrent.TimeUnit;

@Slf4j
@Component
@RequiredArgsConstructor
public class ReservationLockFacade {

    private final RedissonClient redissonClient;
    private final ReservationService reservationService;
    private final ReservationRepository reservationRepository;

    public void reserveRoomWithLock(Long memberId, Long roomId, LocalDate checkIn, LocalDate checkOut, Integer totalPrice) {
        // 객실 고유 ID를 기반으로 락 키 생성 (해당 객실에 대해서만 줄을 세움)
        String lockKey = "room_lock:" + roomId;
        RLock lock = redissonClient.getLock(lockKey);

        try {
            // 락 획득 시도: 최대 5초 대기, 락 획득 후 3초간 유지
            boolean isLocked = lock.tryLock(5, 3, TimeUnit.SECONDS);

            if (!isLocked) {
                log.warn("락 획득 실패 - roomId: {}", roomId);
                throw new RuntimeException("현재 예약자가 많아 처리가 지연되고 있습니다. 잠시 후 다시 시도해주세요.");
            }

            // 💡 락 획득 성공 시, 안전하게 트랜잭션(예약 및 재고 차감) 실행
            reservationService.createReservation(memberId, roomId, checkIn, checkOut, totalPrice);

        } catch (InterruptedException e) {
            log.error("예약 락 대기 중 인터럽트 발생", e);
            throw new RuntimeException("예약 처리 중 시스템 오류가 발생했습니다.");
        } finally {
            // 로직이 끝나면 반드시 락 반납
            if (lock.isLocked() && lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }
    public void cancelReservationWithLock(Long reservationId, Long memberId) {
        // 락을 걸 roomId를 알아내기 위해 예약 정보 먼저 조회
        org.example.hotelbooking.domain.reservation.entity.Reservation reservation =
                reservationRepository.findById(reservationId)
                        .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 예약입니다."));

        Long roomId = reservation.getRoom().getRoomId();
        String lockKey = "room_lock:" + roomId;
        RLock lock = redissonClient.getLock(lockKey);

        try {
            boolean isLocked = lock.tryLock(5, 3, TimeUnit.SECONDS);
            if (!isLocked) {
                log.warn("취소 락 획득 실패 - reservationId: {}", reservationId);
                throw new RuntimeException("현재 처리량이 많아 지연되고 있습니다. 잠시 후 다시 시도해주세요.");
            }

            // 💡 락 획득 성공 시, 안전하게 취소 및 재고 복구 트랜잭션 실행
            reservationService.cancelReservation(reservationId, memberId);

        } catch (InterruptedException e) {
            log.error("취소 락 대기 중 인터럽트 발생", e);
            throw new RuntimeException("취소 처리 중 시스템 오류가 발생했습니다.");
        } finally {
            if (lock.isLocked() && lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }
}

package org.example.hotelbooking.domain.reservation.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.member.entity.Member;
import org.example.hotelbooking.domain.member.repository.MemberRepository;
import org.example.hotelbooking.domain.member.service.MailService;
import org.example.hotelbooking.domain.reservation.dto.ReservationDto;
import org.example.hotelbooking.domain.reservation.entity.Reservation;
import org.example.hotelbooking.domain.reservation.repository.ReservationRepository;
import org.example.hotelbooking.domain.review.repository.ReviewRepository;
import org.example.hotelbooking.domain.room.entity.Room;
import org.example.hotelbooking.domain.room.entity.RoomInventory;
import org.example.hotelbooking.domain.room.repository.RoomInventoryRepository;
import org.example.hotelbooking.domain.room.repository.RoomRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.util.List;

@Service
@RequiredArgsConstructor
public class ReservationService {

    private final ReservationRepository reservationRepository;
    private final RoomInventoryRepository roomInventoryRepository;
    private final MemberRepository memberRepository;
    private final RoomRepository roomRepository;
    private final ReviewRepository reviewRepository;
    private final MailService mailService;

    @Transactional
    public void createReservation(Long memberId, Long roomId, LocalDate checkIn, LocalDate checkOut, Integer totalPrice) {
        // 1. 체크인 ~ 체크아웃 전날(-1일)까지의 재고 조회
        List<RoomInventory> inventories = roomInventoryRepository
                .findInventoriesForReservation(roomId, checkIn, checkOut.minusDays(1));

        // 2. 요청한 숙박 일수와 DB에 세팅된 재고 일수가 일치하는지 검증
        long days = java.time.temporal.ChronoUnit.DAYS.between(checkIn, checkOut);
        if (inventories.size() != days) {
            throw new IllegalArgumentException("예약할 수 없는 날짜가 포함되어 있습니다.");
        }

        // 3. 재고 차감 (Dirty Checking으로 자동 UPDATE 쿼리 발생)
        for (RoomInventory inventory : inventories) {
            inventory.decreaseCount(1);
        }

        // 4. 예약 정보 저장
        Member member = memberRepository.findById(memberId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 회원입니다."));
        Room room = roomRepository.findById(roomId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 객실입니다."));

        Reservation reservation = Reservation.builder()
                .member(member)
                .room(room)
                .checkInDate(checkIn)
                .checkOutDate(checkOut)
                .totalPrice(totalPrice)
                .status("COMPLETED")
                .build();

        Reservation savedReservation = reservationRepository.save(reservation);

        // 💡 5. 예약 확정 안내 이메일 비동기 발송
        Accommodation accom = room.getAccommodation();
        mailService.sendReservationConfirmation(
                member.getEmail(),
                member.getName(),
                savedReservation.getReservationId(),
                accom.getContentId(),
                accom.getTitle(),
                accom.getAddr(),
                accom.getTel(),
                accom.getCheckinTime(),
                accom.getCheckoutTime(),
                room.getRoomTitle(),
                checkIn,
                checkOut,
                totalPrice
        );
    }
    // 기존 createReservation 아래에 추가
    @Transactional
    public void cancelReservation(Long reservationId, Long requesterMemberId) {
        // 1. 예약 정보 조회
        Reservation reservation = reservationRepository.findById(reservationId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 예약입니다."));

        Long guestMemberId = reservation.getMember().getMemberId();
        Member host = reservation.getRoom().getAccommodation().getHost();
        Long hostMemberId = (host != null) ? host.getMemberId() : null;

        boolean isGuest = guestMemberId.equals(requesterMemberId);
        boolean isHostOwner = hostMemberId != null && hostMemberId.equals(requesterMemberId);

        // 2. 본인의 예약인지 검증 및 취소, 숙소 관리자 예약 취소
        if (!isGuest && !isHostOwner) {
            throw new IllegalArgumentException("예약자 본인 또는 해당 숙소의 관리자만 예약을 취소할 수 있습니다.");
        }

        // 3. 예약 상태 취소로 변경 (Dirty Checking)
        reservation.cancel();

        // 4. 재고 복구
        List<RoomInventory> inventories = roomInventoryRepository.findInventoriesForReservation(
                reservation.getRoom().getRoomId(),
                reservation.getCheckInDate(),
                reservation.getCheckOutDate().minusDays(1)
        );

        for (RoomInventory inventory : inventories) {
            inventory.increaseCount(1);
        }
    }

    // 내 예약 목록 조회
    @Transactional(readOnly = true)
    public List<ReservationDto.MyReservationResponse> getMyReservations(String email) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        List<Reservation> reservations = reservationRepository
                .findByMember_MemberIdOrderByCreatedAtDesc(member.getMemberId());

        return reservations.stream()
                .map(r -> {
                    boolean hasReview = reviewRepository.existsByReservation_ReservationId(r.getReservationId());
                    return ReservationDto.MyReservationResponse.of(r, hasReview);
                })
                .toList();
    }

    // 💡 숙소 관리자(호스트)용 내 숙소에 접수된 예약 목록 조회
    @Transactional(readOnly = true)
    public List<ReservationDto.HostReservationResponse> getHostReservations(String email) {
        Member host = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        List<Reservation> reservations = reservationRepository
                .findReservationsByHostId(host.getMemberId());

        return reservations.stream()
                .map(ReservationDto.HostReservationResponse::from)
                .toList();
    }
}

package org.example.hotelbooking.domain.review.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.member.entity.Member;
import org.example.hotelbooking.domain.member.repository.MemberRepository;
import org.example.hotelbooking.domain.reservation.entity.Reservation;
import org.example.hotelbooking.domain.reservation.repository.ReservationRepository;
import org.example.hotelbooking.domain.review.dto.ReviewDto;
import org.example.hotelbooking.domain.review.entity.Review;
import org.example.hotelbooking.domain.review.repository.ReviewRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class ReviewService {

    private final ReviewRepository reviewRepository;
    private final ReservationRepository reservationRepository;
    private final MemberRepository memberRepository;

    // 1. 특정 숙소의 전체 리뷰 조회 (숙소 상세페이지용)
    public List<ReviewDto.Response> getReviewsByStay(String contentId) {
        return reviewRepository.findByAccommodation_ContentIdOrderByCreatedAtDesc(contentId)
                .stream()
                .map(ReviewDto.Response::from)
                .toList();
    }

    // 2. 고객 리뷰 작성
    @Transactional
    public void createReview(String email, ReviewDto.CreateRequest request) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        Reservation reservation = reservationRepository.findById(request.getReservationId())
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 예약입니다."));

        if (!reservation.getMember().getMemberId().equals(member.getMemberId())) {
            throw new IllegalArgumentException("본인이 예약한 내역에만 리뷰를 작성할 수 있습니다.");
        }

        if ("CANCELED".equals(reservation.getStatus())) {
            throw new IllegalArgumentException("취소된 예약에는 리뷰를 작성할 수 없습니다.");
        }

        if (reviewRepository.existsByReservation_ReservationId(reservation.getReservationId())) {
            throw new IllegalStateException("이미 리뷰를 작성한 예약입니다.");
        }

        Accommodation accom = reservation.getRoom().getAccommodation();

        Review review = Review.builder()
                .member(member)
                .accommodation(accom)
                .reservation(reservation)
                .rating(request.getRating() != null ? request.getRating() : 5)
                .content(request.getContent())
                .build();

        reviewRepository.save(review);
    }

    // 3. 숙소 관리자(HOST) 또는 총괄 관리자(ADMIN)용 리뷰 목록 조회
    public List<ReviewDto.Response> getManageReviews(String email) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        String role = member.getRole().getRoleName();
        List<Review> reviews = "ROLE_ADMIN".equals(role)
                ? reviewRepository.findAllWithDetails()
                : reviewRepository.findAllByHostEmail(email);

        return reviews.stream()
                .map(ReviewDto.Response::from)
                .toList();
    }

    // 4. 답글 등록 및 수정
    @Transactional
    public void saveReply(String email, Long reviewId, String replyContent) {
        if (replyContent == null || replyContent.isBlank()) {
            throw new IllegalArgumentException("답글 내용을 입력해 주세요.");
        }

        Review review = getAuthorizedReview(email, reviewId);
        review.updateReply(replyContent.trim());
    }

    // 5. 답글 삭제
    @Transactional
    public void deleteReply(String email, Long reviewId) {
        Review review = getAuthorizedReview(email, reviewId);
        review.deleteReply();
    }

    private Review getAuthorizedReview(String email, Long reviewId) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        Review review = reviewRepository.findById(reviewId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 리뷰입니다."));

        boolean isAdmin = "ROLE_ADMIN".equals(member.getRole().getRoleName());
        boolean isHostOwner = review.getAccommodation().getHost() != null
                && email.equals(review.getAccommodation().getHost().getEmail());

        if (!isAdmin && !isHostOwner) {
            throw new IllegalArgumentException("해당 숙소의 관리자만 답글을 관리할 수 있습니다.");
        }
        return review;
    }
}

package org.example.hotelbooking.domain.tour.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.accommodation.repository.AccommodationRepository;
import org.example.hotelbooking.domain.tour.dto.NearbyTourResponseDto;
import org.example.hotelbooking.domain.tour.repository.TourAttractionRepository;
import org.springframework.data.geo.Circle;
import org.springframework.data.geo.Distance;
import org.springframework.data.geo.GeoResult;
import org.springframework.data.geo.GeoResults;
import org.springframework.data.geo.Point;
import org.springframework.data.redis.connection.RedisGeoCommands;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.data.redis.domain.geo.Metrics;
import org.springframework.stereotype.Service;

import java.util.ArrayList;
import java.util.List;

@Service
@RequiredArgsConstructor
public class TourSearchService {

    private final StringRedisTemplate redisTemplate;
    private final TourAttractionRepository tourRepository;
    private final AccommodationRepository accommodationRepository;

    public List<NearbyTourResponseDto> getNearbyTours(String accomContentId) {
        // 기준점(해당 숙소)의 좌표를 오라클에서 가져옵니다.
        Accommodation accom = accommodationRepository.findById(accomContentId)
                .orElseThrow(() -> new IllegalArgumentException("숙소를 찾을 수 없습니다."));

        double accomX = Double.parseDouble(accom.getMapX());
        double accomY = Double.parseDouble(accom.getMapY());

        // 2. Redis에 반경 20km 이내의 관광지를 거리순(오름차순)으로 요청합니다.
        String key = "tour:geo:locations";
        Circle circle = new Circle(new Point(accomX, accomY), new Distance(20.0, Metrics.KILOMETERS));

        RedisGeoCommands.GeoRadiusCommandArgs args =
                RedisGeoCommands.GeoRadiusCommandArgs.newGeoRadiusArgs()
                        .includeDistance()
                        .sortAscending()
                        .limit(9); // 최대 9개까지만 노출

        GeoResults<RedisGeoCommands.GeoLocation<String>> geoResults =
                redisTemplate.opsForGeo().radius(key, circle, args);

        List<NearbyTourResponseDto> resultList = new ArrayList<>();

        if (geoResults != null) {
            for (GeoResult<RedisGeoCommands.GeoLocation<String>> geoResult : geoResults) {
                // Redis에서 가져온 관광지 ID와 거리
                String tourId = geoResult.getContent().getName();
                double distance = geoResult.getDistance().getValue();

                // 관광지 ID로 오라클 DB에서 이름, 사진 등을 가져와 DTO로 합칩니다.
                tourRepository.findById(tourId).ifPresent(tour -> {
                    double formattedDistance = Math.round(distance * 10) / 10.0;

                    resultList.add(new NearbyTourResponseDto(
                            tour.getContentId(),
                            tour.getTitle(),
                            tour.getFirstImage(),
                            formattedDistance
                    ));
                });
            }
        }
        return resultList;
    }
}

package org.example.hotelbooking.domain.tour.service;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.example.hotelbooking.domain.tour.entity.TourAttraction;
import org.example.hotelbooking.domain.tour.repository.TourAttractionRepository;
import org.springframework.boot.context.event.ApplicationReadyEvent;
import org.springframework.context.event.EventListener;
import org.springframework.data.geo.Point;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Component;

import java.util.List;

@Component
@RequiredArgsConstructor
@Slf4j
public class TourGeoInitializer {

    private final StringRedisTemplate redisTemplate;
    private final TourAttractionRepository tourRepository;

    @EventListener(ApplicationReadyEvent.class)
    public void syncTourCoordinatesToRedis() {
        String key = "tour:geo:locations";

        // 기존에 남아있던 캐시 삭제 (초기화)
        redisTemplate.delete(key);

        List<TourAttraction> tours = tourRepository.findAll();

        for (TourAttraction tour : tours) {
            try {
                // 좌표값이 비어있지 않은 데이터만 Redis에 저장
                if (tour.getMapX() != null && tour.getMapY() != null) {
                    double x = Double.parseDouble(tour.getMapX());
                    double y = Double.parseDouble(tour.getMapY());

                    redisTemplate.opsForGeo().add(key, new Point(x, y), tour.getContentId());
                }
            } catch (NumberFormatException e) {
                log.warn("좌표 변환 실패 - 관광지 ID: {}", tour.getContentId());
            }
        }
        log.info("✅ Redis에 {}개의 관광지 좌표 적재 완료!", tours.size());
    }
}

package org.example.hotelbooking.domain.tour.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.member.entity.Member;
import org.example.hotelbooking.domain.member.repository.MemberRepository;
import org.example.hotelbooking.domain.tour.dto.TourDto;
import org.example.hotelbooking.domain.tour.entity.TourAttraction;
import org.example.hotelbooking.domain.tour.entity.TourImage;
import org.example.hotelbooking.domain.tour.entity.TourLike;
import org.example.hotelbooking.domain.tour.repository.TourAttractionRepository;
import org.example.hotelbooking.domain.tour.repository.TourImageRepository;
import org.example.hotelbooking.domain.tour.repository.TourLikeRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;
import java.util.Optional;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class TourService {

    private final TourAttractionRepository tourAttractionRepository;
    private final TourImageRepository tourImageRepository;
    private final TourLikeRepository tourLikeRepository;
    private final MemberRepository memberRepository;

    // 1. 전체 관광지 목록 조회 (지도 표시용)
    public List<TourDto.SummaryResponse> getAllTours() {
        return tourAttractionRepository.findAllByOrderByTitleAsc()
                .stream()
                .map(TourDto.SummaryResponse::from)
                .toList();
    }

    // 2. 관광지 단건 상세 조회
    public TourDto.DetailResponse getTourDetail(String contentId) {
        TourAttraction tour = tourAttractionRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("관광지 정보를 찾을 수 없습니다. contentId=" + contentId));

        List<TourImage> images = tourImageRepository.findByTourAttraction_ContentIdOrderByImageIdAsc(contentId);
        return TourDto.DetailResponse.of(tour, images);
    }

    // 3. 관광지 찜 여부 확인
    public boolean isLiked(String email, String contentId) {
        if (email == null || contentId == null) return false;
        Member member = memberRepository.findByEmail(email).orElse(null);
        if (member == null) return false;
        return tourLikeRepository.existsByMember_MemberIdAndTourAttraction_ContentId(member.getMemberId(), contentId);
    }

    // 4. 관광지 찜 토글
    @Transactional
    public boolean toggleTourLike(String email, String contentId) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        TourAttraction tour = tourAttractionRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("관광지를 찾을 수 없습니다."));

        Optional<TourLike> existing = tourLikeRepository
                .findByMember_MemberIdAndTourAttraction_ContentId(member.getMemberId(), contentId);

        if (existing.isPresent()) {
            tourLikeRepository.delete(existing.get());
            return false;
        } else {
            tourLikeRepository.save(TourLike.builder()
                    .member(member)
                    .tourAttraction(tour)
                    .build());
            return true;
        }
    }

    // 5. 내가 찜한 관광지 목록 조회
    public List<TourDto.SummaryResponse> getMyLikedTours(String email) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        return tourLikeRepository.findByMemberIdWithTour(member.getMemberId())
                .stream()
                .map(tl -> TourDto.SummaryResponse.from(tl.getTourAttraction()))
                .toList();
    }
}

package org.example.hotelbooking.domain.accommodation.repository;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

import java.time.LocalDate;
import java.util.List;

public interface AccommodationRepository extends JpaRepository<Accommodation, String>, AccommodationRepositoryCustom {

    @Query("SELECT new org.example.hotelbooking.domain.accommodation.dto.StayResponseDto(" +
            "a.contentId, a.host.memberId, a.title, a.lodgingType, a.tel, a.zipcode, a.addr, " +
            "a.checkinTime, a.checkoutTime, a.roomCount, a.roomType, a.parking, a.subFacility, " +
            "a.overview, a.firstImage, a.firstImage2, a.mapX, a.mapY, " +
            "COALESCE((SELECT MIN(r.pricePerNight) FROM Room r WHERE r.accommodation = a), 0), " +
            "COALESCE((SELECT SUM(ri.availableCount) FROM RoomInventory ri JOIN ri.room r " +
            "          WHERE r.accommodation = a " +
            "            AND ri.inventoryDate = :checkIn " +
            "            AND (:guests IS NULL OR :guests <= 0 OR CAST(COALESCE(r.roomMaxCount, '0') AS integer) >= :guests)), 0L)) " +
            "FROM Accommodation a " +
            "WHERE a.approvalStatus = org.example.hotelbooking.domain.accommodation.entity.AccommodationStatus.APPROVED " +
            "  AND (:region IS NULL OR :region = '' OR :region = '전체' OR a.addr LIKE %:region%) " +
            "  AND (:type IS NULL OR :type = '' OR :type = '전체' OR a.lodgingType LIKE %:type%) " +
            "  AND (:query IS NULL OR :query = '' OR a.title LIKE %:query% OR a.addr LIKE %:query%) " +
            "  AND (:guests IS NULL OR :guests <= 0 OR EXISTS (" +
            "       SELECT 1 FROM Room r " +
            "       WHERE r.accommodation = a " +
            "         AND CAST(COALESCE(r.roomMaxCount, '0') AS integer) >= :guests)) " +
            // 💡 날짜가 없는 전체 조회일 때는 통과, 날짜가 들어왔을 때는 :stayDays > 0 조건과 일자별 잔여 재고를 엄격히 검증
            "  AND (:stayDays = 0L OR EXISTS (" +
            "           SELECT 1 FROM Room r " +
            "           WHERE r.accommodation = a " +
            "             AND (SELECT COUNT(ri) FROM RoomInventory ri " +
            "                  WHERE ri.room = r " +
            "                    AND ri.inventoryDate >= :checkIn " +
            "                    AND ri.inventoryDate < :checkOut " +
            "                    AND ri.availableCount > 0) = :stayDays)) " +
            "ORDER BY a.contentId DESC")
    List<StayResponseDto> searchStays(
            @Param("region") String region,
            @Param("type") String type,
            @Param("query") String query,
            @Param("checkIn") LocalDate checkIn,
            @Param("checkOut") LocalDate checkOut,
            @Param("stayDays") Long stayDays,
            @Param("guests") Integer guests // 💡 [추가]
    );

    // 숙소 생성시 시퀀스 생성
    @Query(value = "SELECT SEQ_HOST_ACCOM.NEXTVAL FROM DUAL", nativeQuery = true)
    Long getNextHostAccomSeq();

    // 호스트가 등록한 숙소 찾아오는 jpa쿼리
    List<Accommodation> findByHost_Email(String email);

    // 숙소 관리자의 숙소 보유 여부 확인
    boolean existsByHost_MemberIdAndApprovalStatus(Long memberId, org.example.hotelbooking.domain.accommodation.entity.AccommodationStatus approvalStatus);

    // 특정 호스트의 반려된 숙소 목록 조회 (반려 알림용)
    List<Accommodation> findByHost_MemberIdAndApprovalStatus(Long memberId, AccommodationStatus approvalStatus);

    @Column(name = "ROOM_COUNT", length = 50)

    @Lob
    private String overview;

    @Column(name = "FIRST_IMAGE", length = 1000)
    private String firstImage;

    @Column(name = "FIRST_IMAGE2", length = 1000)
    private String firstImage2;

    @Column(name = "MAP_X", length = 50)

    @Column(name = "MAP_Y", length = 50)

    @Enumerated(EnumType.STRING)
    @Column(name = "APPROVAL_STATUS", length = 20, nullable = false)
    private AccommodationStatus approvalStatus;

    @Column(name = "REJECT_REASON", length = 255)
    private String rejectReason;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "MEMBER_ID")
    private Member host;

    @Column(name = "CREATED_AT", updatable = false)
    private LocalDateTime createdAt;

    @PrePersist
    protected void onCreate() {
        this.createdAt = LocalDateTime.now();
        if (this.approvalStatus == null) {
            this.approvalStatus = AccommodationStatus.PENDING;
        }
    }

        this.approvalStatus = AccommodationStatus.APPROVED;
        this.rejectReason = null;

        this.rejectReason = reason;
    }

        this.rejectReason = rejectReason;
    }

    // 💡 호스트의 숙소 정보 수정 편의 메서드 (Accommodation.java 내부에 추가)

package org.example.hotelbooking.domain.accommodation.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.dto.StayDetailResponseDto;
import org.example.hotelbooking.domain.accommodation.dto.StayResponseDto;

import org.example.hotelbooking.domain.accommodation.repository.AccommodationRepository;

import org.example.hotelbooking.domain.room.entity.RoomImage;
import org.example.hotelbooking.domain.room.entity.RoomInventory;
import org.example.hotelbooking.domain.room.repository.RoomImageRepository;
import org.example.hotelbooking.domain.room.repository.RoomInventoryRepository;
import org.example.hotelbooking.domain.room.repository.RoomRepository;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.time.temporal.ChronoUnit;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class StayService {

    private final AccommodationRepository accommodationRepository;
    private final RoomRepository roomRepository;
    private final RoomImageRepository roomImageRepository;
    private final RoomInventoryRepository roomInventoryRepository; // 💡 재고 조회용 주입

    @Cacheable(value = "stayListCache", key = "'allStays'")
    public List<StayResponseDto> getStays(String region, String type, String query, LocalDate checkIn, LocalDate checkOut, Integer guests) {
        Long stayDays = 0L;

        if (checkIn != null && checkOut != null && checkOut.isAfter(checkIn)) {
            stayDays = ChronoUnit.DAYS.between(checkIn, checkOut);
        } else {
            checkIn = LocalDate.now();
            checkOut = LocalDate.now();
        }

        return accommodationRepository.searchStays(region, type, query, checkIn, checkOut, stayDays, guests);
    }

    // 💡 [수정] 체크인/체크아웃 날짜를 받아 객실별 잔여 개수 계산
    public StayDetailResponseDto getStayDetail(String contentId, LocalDate checkIn, LocalDate checkOut) {
        Accommodation accommodation = accommodationRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("숙소를 찾을 수 없습니다. contentId=" + contentId));

        // 날짜가 지정되지 않았으면 기본 오늘 체크인 ~ 내일 체크아웃(1박) 기준
        if (checkIn == null) {
            checkIn = LocalDate.now();
        }
        if (checkOut == null || !checkOut.isAfter(checkIn)) {
            checkOut = checkIn.plusDays(1);
        }

        long stayDays = ChronoUnit.DAYS.between(checkIn, checkOut);

        List<Room> rooms = roomRepository.findByAccommodation_ContentId(contentId);

        // 해당 숙소의 기간 내 모든 재고 조회 후 roomId별로 그룹핑
        List<RoomInventory> inventories = roomInventoryRepository.findByContentIdAndDateRange(contentId, checkIn, checkOut);
        Map<Long, List<RoomInventory>> invByRoomMap = inventories.stream()
                .collect(Collectors.groupingBy(ri -> ri.getRoom().getRoomId()));

        Map<Long, List<RoomImage>> roomImagesMap = new HashMap<>();
        Map<Long, Integer> roomAvailableMap = new HashMap<>();

        for (Room room : rooms) {
            List<RoomImage> images = roomImageRepository.findByRoom_RoomIdOrderByDisplayOrderAsc(room.getRoomId());
            roomImagesMap.put(room.getRoomId(), images);

            List<RoomInventory> roomInvs = invByRoomMap.getOrDefault(room.getRoomId(), List.of());
            // 숙박 일수(stayDays)만큼 모든 날짜에 재고 데이터가 있어야 하며, 그중 최소 재고를 잔여 객실 수로 설정
            if (roomInvs.size() < stayDays) {
                roomAvailableMap.put(room.getRoomId(), 0);
            } else {
                int minAvailable = roomInvs.stream()
                        .mapToInt(RoomInventory::getAvailableCount)
                        .min()
                        .orElse(0);
                roomAvailableMap.put(room.getRoomId(), Math.max(minAvailable, 0));
            }
        }

        return StayDetailResponseDto.of(accommodation, rooms, roomImagesMap, roomAvailableMap);
    }
}

package org.example.hotelbooking.domain.accommodation.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.dto.AccommodationDto;
import org.example.hotelbooking.domain.accommodation.dto.AccommodationResponse;

import java.time.LocalDate;
import java.util.List;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true)
public class AccommodationService {

    private final AccommodationRepository accommodationRepository;
    private final MemberRepository memberRepository;
    private final ReservationRepository reservationRepository;
    private final RoomRepository roomRepository;
    private final RoomImageRepository roomImageRepository;
    private final RoomInventoryRepository roomInventoryRepository;
    private final AccomLikeRepository accomLikeRepository;
    private final ReviewRepository reviewRepository;

    // 지도 기반 숙소 조회
    public List<AccommodationResponse> getAccommodationsInMap(String minX, String maxX, String minY, String maxY) {
        return accommodationRepository.findAccommodationsInMap(minX, maxX, minY, maxY)
                .stream()
                .map(AccommodationResponse::from)
                .collect(Collectors.toList());
    }

    // 💡 1. 숙소 신규 등록 (이미지 및 좌표 컬럼까지 모두 저장)
    @Transactional
    public String register(String email, AccommodationDto.RegisterRequest request) {
        Member host = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자 정보를 찾을 수 없습니다."));

        String contentId = request.getContentId();
        if (contentId == null || contentId.isBlank()) {
            Long seqNumber = accommodationRepository.getNextHostAccomSeq();
            contentId = String.valueOf(seqNumber);
        }

        Accommodation accommodation = Accommodation.builder()
                .contentId(contentId)
                .title(request.getTitle())
                .lodgingType(request.getLodgingType())
                .tel(request.getTel())
                .zipcode(request.getZipcode())
                .addr(request.getAddr())
                .checkinTime(request.getCheckinTime())
                .checkoutTime(request.getCheckoutTime())
                .roomCount(request.getRoomCount())
                .roomType(request.getRoomType())
                .parking(request.getParking())
                .subFacility(request.getSubFacility())
                .overview(request.getOverview())
                .firstImage(request.getFirstImage())
                .firstImage2(request.getFirstImage2())
                .mapX(request.getMapX())
                .mapY(request.getMapY())
                .host(host)
                .approvalStatus(AccommodationStatus.PENDING)
                .build();

        accommodationRepository.save(accommodation);
        return accommodation.getContentId();
    }

    // 💡 2. 호스트 내 숙소 전체 컬럼 조회 (수정 폼 바인딩용)
    public List<AccommodationDto.HostDetailResponse> getMyAccommodations(String email) {
        return accommodationRepository.findByHost_Email(email)
                .stream()
                .map(AccommodationDto.HostDetailResponse::from)
                .collect(Collectors.toList());
    }

    // 💡 3. 호스트 내 숙소 정보 수정 (Dirty Checking으로 DB 업데이트)
    @Transactional
    public void updateMyAccommodation(String email, String contentId, AccommodationDto.UpdateRequest request) {
        Accommodation accommodation = accommodationRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 숙소입니다. id=" + contentId));

        if (accommodation.getHost() == null || !accommodation.getHost().getEmail().equals(email)) {
            throw new IllegalArgumentException("본인이 등록한 숙소만 수정할 수 있습니다.");
        }

        accommodation.updateDetails(
                request.getTitle(),
                request.getLodgingType(),
                request.getTel(),
                request.getZipcode(),
                request.getAddr(),
                request.getCheckinTime(),
                request.getCheckoutTime(),
                request.getRoomCount(),
                request.getRoomType(),
                request.getParking(),
                request.getSubFacility(),
                request.getOverview(),
                request.getFirstImage(),
                request.getFirstImage2(),
                request.getMapX(),
                request.getMapY()
        );
    }

    // 💡 4. [신규 추가] 호스트 내 숙소 삭제 (진행 예정인 예약이 없을 때만 허용)
    // 💡 4. [벌크 연산 튜닝 적용 완료 🚀] 호스트 내 숙소 삭제 (진행 예정인 예약이 없을 때만 허용)
    @Transactional
    public void deleteMyAccommodation(String email, String contentId) {
        Accommodation accommodation = accommodationRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 숙소입니다. id=" + contentId));

        if (accommodation.getHost() == null || !accommodation.getHost().getEmail().equals(email)) {
            throw new IllegalArgumentException("본인이 등록한 숙소만 삭제할 수 있습니다.");
        }

        Long hostMemberId = accommodation.getHost().getMemberId();
        LocalDate today = LocalDate.now();

        // 1. 해당 숙소에 속한 모든 예약 조회 후 투숙 예정 검증
        List<Reservation> targetReservations = reservationRepository.findReservationsByHostId(hostMemberId)
                .stream()
                .filter(r -> contentId.equals(r.getRoom().getAccommodation().getContentId()))
                .toList();

        boolean hasActiveReservation = targetReservations.stream()
                .anyMatch(r -> !"CANCELED".equals(r.getStatus()) && r.getCheckOutDate().isAfter(today));

        if (hasActiveReservation) {
            throw new IllegalStateException("해당 숙소에 진행 중이거나 예정된 예약이 존재하여 삭제할 수 없습니다.");
        }

        // --- 🚀 [성능 최적화: JPQL Bulk 연산을 통한 즉각 삭제 처리] ---

        // 2. 리뷰 및 찜(위시리스트) 일괄 삭제
        reviewRepository.deleteByAccommodation_ContentId(contentId);
        accomLikeRepository.deleteByAccommodation_ContentId(contentId);

        // 3. 과거 완료/취소된 예약 내역 일괄 삭제 (JPA deleteAll 묶음 쿼리 활용)
        if (!targetReservations.isEmpty()) {
            reservationRepository.deleteAllInBatch(targetReservations);
        }

        // 4. 🔥 수천 건의 객실 재고(90일 치)와 사진을 단 2개의 쿼리로 한 방에 삭제! (for문 제거)
        roomInventoryRepository.deleteAllInventoriesByAccommodationIdBulk(contentId);
        roomImageRepository.deleteAllImagesByAccommodationIdBulk(contentId);

        // 5. 숙소에 딸린 객실 껍데기(Room) 일괄 삭제 (벌크 연산)
        List<Room> rooms = roomRepository.findByAccommodation_ContentId(contentId);
        if (!rooms.isEmpty()) {
            roomRepository.deleteAllInBatch(rooms);
        }

        // 6. 최종적으로 부모 객체인 숙소(ACCOMMODATION) 삭제
        accommodationRepository.delete(accommodation);
    }

    // 호스트의 숙소 보유 여부 확인 (PENDING / APPROVED 등 등록된 숙소가 1개라도 있는지 여부 또는 승인 숙소 여부)
    public boolean hasApprovedAccommodation(String email) {
        Member host = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자 정보를 찾을 수 없습니다."));

        return accommodationRepository.existsByHost_MemberIdAndApprovalStatus(host.getMemberId(), AccommodationStatus.APPROVED);
    }

    // 💡 [총괄관리자용] 승인 대기(PENDING) 숙소 목록 최신순 상세 조회
    public List<AccommodationDto.HostDetailResponse> getPendingAccommodations() {
        return accommodationRepository.findByApprovalStatusOrderByCreatedAtDesc(AccommodationStatus.PENDING)
                .stream()
                .map(AccommodationDto.HostDetailResponse::from)
                .collect(Collectors.toList());
    }

    // [총괄관리자용] 숙소 승인 처리
    @Transactional
    public void approveAccommodation(String contentId) {
        Accommodation accommodation = accommodationRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 숙소입니다. id=" + contentId));
        accommodation.approve();
    }

    // [총괄관리자용] 숙소 반려 처리
    @Transactional
    public void rejectAccommodation(String contentId, String rejectReason) {
        if (rejectReason == null || rejectReason.isBlank()) {
            throw new IllegalArgumentException("반려 사유를 입력해야 합니다.");
        }
        Accommodation accommodation = accommodationRepository.findById(contentId)
                .orElseThrow(() -> new IllegalArgumentException("존재하지 않는 숙소입니다. id=" + contentId));
        accommodation.reject(rejectReason);
    }
}

import lombok.RequiredArgsConstructor;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.Customizer;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;
import org.springframework.web.cors.CorsConfiguration;
import org.springframework.web.cors.CorsConfigurationSource;
import org.springframework.web.cors.UrlBasedCorsConfigurationSource;

import java.util.Arrays;
import java.util.List;

@Configuration
@EnableWebSecurity
@RequiredArgsConstructor
public class SecurityConfig {

    private final JwtTokenProvider jwtTokenProvider;

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
                // React 통신을 위한 CORS 설정
                .cors(Customizer.withDefaults())

                // REST API이므로 CSRF 보호 비활성화
                .csrf(csrf -> csrf.disable())

                // JWT를 사용하므로 서버 세션을 사용하지 않음
                .sessionManagement(session ->
                        session.sessionCreationPolicy(SessionCreationPolicy.STATELESS)
                )

                // API 및 정적 리소스 접근 권한 설정
                .authorizeHttpRequests(auth -> auth
                        // 1. React 정적 리소스 전체 허용 (Vite 빌드 파일 및 루트)
                        .requestMatchers(
                                "/",
                                "/index.html",
                                "/assets/**",
                                "/vite.svg",
                                "/oauth/callback",
                                "/*.ico",
                                "/*.png",
                                "/*.jpg",
                                "/*.jpeg",
                                "/*.svg",
                                "/*.json",
                                "/*.js",
                                "/*.css"
                        ).permitAll()

                        // 💡 스프링 부트 기본 에러 페이지 포워딩 시 403 마스킹 방지
                        .requestMatchers("/error").permitAll()

                        // 로그인 / 회원가입
                        .requestMatchers("/api/v1/auth/**").permitAll()

                        // 숙소 조회
                        .requestMatchers("/api/v1/accommodations/**").permitAll()

                        // 숙소 조회 API
                        .requestMatchers("/api/v1/stays/**").permitAll()

                        // 관광지 조회 api
                        .requestMatchers("/api/v1/tours/**").permitAll()

                        // 비밀번호 찾기 위한 허용 설정
                        .requestMatchers("/api/v1/members/password/**").permitAll()

                        // 토스페이먼츠 클라이언트 키 조회 허용
                        .requestMatchers("/api/v1/payments/client-key").permitAll()

                        // 숙소 리뷰 조회 허용
                        .requestMatchers(org.springframework.http.HttpMethod.GET, "/api/v1/reviews/stays/**").permitAll()

                        // 그 외 요청은 JWT 인증 필요
                        .anyRequest().authenticated()
                )

                // JWT 인증 필터 등록
                .addFilterBefore(
                        new JwtAuthenticationFilter(jwtTokenProvider),
                        UsernamePasswordAuthenticationFilter.class
                );

        return http.build();
    }

    // 비밀번호 암호화
    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    // CORS 설정
    @Bean
    public CorsConfigurationSource corsConfigurationSource() {
        CorsConfiguration configuration = new CorsConfiguration();

        // 로컬 개발 환경(Vite) + 배포 서버 IP 주소 허용
        configuration.setAllowedOrigins(
                List.of(
                        "http://localhost:5173",
                        "http://150.230.7.97:8080"
                )
        );

        configuration.setAllowedMethods(
                Arrays.asList(
                        "GET",
                        "POST",
                        "PUT",
                        "DELETE",
                        "OPTIONS",
                        "PATCH"
                )
        );

        configuration.setAllowedHeaders(List.of("*"));
        configuration.setAllowCredentials(true);
        configuration.setMaxAge(3600L);

        UrlBasedCorsConfigurationSource source =
                new UrlBasedCorsConfigurationSource();

package org.example.hotelbooking.global.security;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.util.StringUtils;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.Collections;

// 💡 롬복을 써서 JwtTokenProvider를 자동으로 주입받습니다.
@RequiredArgsConstructor
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private final JwtTokenProvider jwtTokenProvider;

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
            throws ServletException, IOException {

        // 1. 사용자 요청 헤더에서 JWT 토큰을 쓱 뽑아옵니다. (아래 만들어둔 resolveToken 메서드 사용)
        String token = resolveToken(request);

        // 2. 토큰이 비어있지 않고, 진품(유효기간 안 지남 등)인지 검사합니다.
        if (token != null && jwtTokenProvider.validateToken(token)) {

            // 3. 진품이라면 토큰 안에 적혀있는 사용자 이메일(Subject)과 역할(Role)을 꺼냅니다.
            String email = jwtTokenProvider.getUserEmail(token);
            String role = jwtTokenProvider.getUserRole(token);

            // 4. 꺼낸 정보를 바탕으로 "이 사람은 인증된 사용자입니다" 라는 확인증(Authentication 객체)을 만듭니다.
            Authentication authentication = new UsernamePasswordAuthenticationToken(
                    email,
                    null,
                    Collections.singletonList(new SimpleGrantedAuthority(role))
            );

            // 5. 스프링 시큐리티의 핵심 저장소(SecurityContext)에 이 확인증을 예쁘게 보관해 둡니다.
            //    (이렇게 해두면 나중에 컨트롤러에서 @AuthenticationPrincipal 로 이 사람 정보를 꺼내 쓸 수 있습니다!)
            SecurityContextHolder.getContext().setAuthentication(authentication);
        }

        // 6. 검사가 끝났으니 다음 필터나 컨트롤러로 무사히 넘겨줍니다.
        filterChain.doFilter(request, response);
    }

    // 💡 프론트엔드가 보낸 요청(Header) 주머니에서 "Bearer xxxx.yyyy.zzzz" 형태의 토큰만 쏙 빼내는 기능
    private String resolveToken(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        // 토큰이 존재하고, "Bearer "로 시작하면 (7글자)
        if (StringUtils.hasText(bearerToken) && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7); // "Bearer " 이후의 진짜 토큰 부분만 잘라서 반환
        }
        return null; // 토큰이 없으면 null 반환
    }
}

package org.example.hotelbooking.global.security;

import io.jsonwebtoken.*;
import io.jsonwebtoken.io.Decoders;
import io.jsonwebtoken.security.Keys;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.security.Key;
import java.util.Date;

@Slf4j
@Component
public class JwtTokenProvider {

    private final Key key;
    private final long expirationTime;

    // yml 또는 properties 파일의 값을 불러와서 Key 객체로 만듭니다.
    public JwtTokenProvider(@Value("${jwt.secret}") String secretKey,
                            @Value("${jwt.expiration}") long expirationTime) {
        byte[] keyBytes = Decoders.BASE64.decode(secretKey);
        this.key = Keys.hmacShaKeyFor(keyBytes);
        this.expirationTime = expirationTime;
    }

    // 💡 1. 토큰 생성 (로그인 성공 시 프론트엔드로 내려줄 출입증)
    // 프론트엔드에서 로그인 시 실제 닉네임을 띄우기 위해 String name, .claim("name", name) 추가
    public String createToken(String email, String role, String name) {
        Date now = new Date();
        Date validity = new Date(now.getTime() + this.expirationTime);

        return Jwts.builder()
                .setSubject(email) // 토큰 주인을 이메일로 기록
                .claim("role", role) // 권한 정보 기록 (예: ROLE_USER)
                .claim("name", name) // 💡 토큰 claim에 DB name 추가
                .setIssuedAt(now) // 발행 시간
                .setExpiration(validity) // 만료 시간
                .signWith(key, SignatureAlgorithm.HS256) // 암호화 알고리즘
                .compact();
    }
    // 💡수명이 7일 인 Refresh Token 생성
    public String createRefreshToken(String email) {
        Date now = new Date();
        Date validity = new Date(now.getTime() + 604800000L); // 7일

        return Jwts.builder()
                .setSubject(email)
                .setIssuedAt(now)
                .setExpiration(validity)
                .signWith(key, SignatureAlgorithm.HS256)
                .compact();
    }
    // 💡 2. 토큰 유효성 검증 (API 요청이 올 때마다 진짜 출입증인지 확인)
    public boolean validateToken(String token) {
        try {
            Jwts.parserBuilder().setSigningKey(key).build().parseClaimsJws(token);
            return true;
        } catch (io.jsonwebtoken.security.SecurityException | MalformedJwtException e) {
            log.info("잘못된 JWT 서명입니다.");
        } catch (ExpiredJwtException e) {
            log.info("만료된 JWT 토큰입니다.");
        } catch (UnsupportedJwtException e) {
            log.info("지원되지 않는 JWT 토큰입니다.");
        } catch (IllegalArgumentException e) {
            log.info("JWT 토큰이 잘못되었습니다.");
        }
        return false;

    }
    // 💡 3. 토큰을 분해해서 주인의 이메일(Subject)을 꺼내는 기능
    public String getUserEmail(String token) {
        return Jwts.parserBuilder().setSigningKey(key).build()
                .parseClaimsJws(token)
                .getBody()
                .getSubject();
    }

    // 💡 4. 토큰을 분해해서 주인의 권한(Role)을 꺼내는 기능
    public String getUserRole(String token) {
        return Jwts.parserBuilder().setSigningKey(key).build()
                .parseClaimsJws(token)
                .getBody()
                .get("role", String.class);
    }
}
