package org.example.hotelbooking.domain.accommodation.controller;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.dto.AccommodationDto;
import org.example.hotelbooking.domain.accommodation.dto.AccommodationResponse;
import org.example.hotelbooking.domain.accommodation.service.AccommodationService;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/v1/accommodations")
@RequiredArgsConstructor
public class AccommodationController {

    private final AccommodationService accommodationService;

    // 기존에 만드신 지도 기반 숙소 조회 API (유지)
    @GetMapping
    public ResponseEntity<List<AccommodationResponse>> getMapAccommodations(
            @RequestParam String minX,
            @RequestParam String maxX,
            @RequestParam String minY,
            @RequestParam String maxY) {

        List<AccommodationResponse> response = accommodationService.getAccommodationsInMap(minX, maxX, minY, maxY);
        return ResponseEntity.ok(response);
    }

    // 💡 새롭게 추가하는 숙소 등록 API
    @PreAuthorize("hasRole('HOST')")
    @PostMapping("/register")
    public ResponseEntity<String> register(
            @RequestBody AccommodationDto.RegisterRequest request,
            @AuthenticationPrincipal String email // JWT 필터에서 넣어둔 유저 이메일
    ) {
        String contentId = accommodationService.register(email, request);
        return ResponseEntity.ok("숙소 등록 완료! 숙소 ID: " + contentId);
    }
    @PreAuthorize("hasRole('HOST')")
    @GetMapping("/my")
    public ResponseEntity<List<AccommodationResponse>> getMyAccommodations(
            // 혹시 CustomUserDetails 같은 객체로 꺼내야 한다면 여기서 에러가 나거나 null일 수 있습니다.
            @AuthenticationPrincipal String email
    ) {
        // 💡 콘솔창에 이메일이 정상적으로 찍히는지 확인하는 코드!
        System.out.println("토큰에서 꺼낸 이메일: " + email);

        List<AccommodationResponse> myAccommodations = accommodationService.getMyAccommodations(email);
        return ResponseEntity.ok(myAccommodations);
    }

    // 💡 승인된 숙소 보유 여부 확인 API
    @GetMapping("/has-approved")
    public ResponseEntity<Boolean> checkApprovedAccommodation(@AuthenticationPrincipal String email) {
        if (email == null) {
            return ResponseEntity.ok(false);
        }
        boolean hasApproved = accommodationService.hasApprovedAccommodation(email);
        return ResponseEntity.ok(hasApproved);
    }
}

package org.example.hotelbooking.domain.accommodation.entity;

import org.example.hotelbooking.domain.member.entity.Member;
import jakarta.persistence.*;
import lombok.*;
import java.time.LocalDateTime;

@Entity
@Getter
@NoArgsConstructor(access = AccessLevel.PROTECTED)
@Table(name = "ACCOMMODATION")
public class Accommodation {

    @Id
    @Column(name = "CONTENT_ID", length = 20)
    private String contentId;

    @Column(nullable = false, length = 200)
    private String title;

    @Column(name = "LODGING_TYPE", length = 50)
    private String lodgingType;

    @Column(length = 100)
    private String tel;

    @Column(length = 10)
    private String zipcode;

    @Column(length = 300)
    private String addr;

    @Column(name = "CHECKIN_TIME", length = 100)
    private String checkinTime;

    @Column(name = "CHECKOUT_TIME", length = 100)
    private String checkoutTime;

    @Column(name = "ROOM_COUNT", length = 50)
    private String roomCount;

    @Column(name = "ROOM_TYPE", length = 200)
    private String roomType;

    @Column(length = 300)
    private String parking;

    @Column(name = "SUB_FACILITY", length = 1000)
    private String subFacility;

    @Lob
    private String overview;

    @Column(name = "FIRST_IMAGE", length = 1000)
    private String firstImage;

    @Column(name = "FIRST_IMAGE2", length = 1000)
    private String firstImage2;

    @Column(name = "MAP_X", length = 50)
    private String mapX;

    @Column(name = "MAP_Y", length = 50)
    private String mapY;

    // 기존 APPROVAL_STATUS 컬럼을 Enum(PENDING, APPROVED, REJECTED)으로 매핑
    @Enumerated(EnumType.STRING)
    @Column(name = "APPROVAL_STATUS", length = 20, nullable = false)
    private AccommodationStatus approvalStatus;

    @Column(name = "REJECT_REASON", length = 255)
    private String rejectReason;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "MEMBER_ID")
    private Member host;

    @Column(name = "CREATED_AT", updatable = false)
    private LocalDateTime createdAt;

    @PrePersist
    protected void onCreate() {
        this.createdAt = LocalDateTime.now();
        if (this.approvalStatus == null) {
            this.approvalStatus = AccommodationStatus.PENDING;
        }
    }

    // 관리자 승인/반려 편의 메서드
    public void approve() {
        this.approvalStatus = AccommodationStatus.APPROVED;
        this.rejectReason = null;
    }

    public void reject(String reason) {
        this.approvalStatus = AccommodationStatus.REJECTED;
        this.rejectReason = reason;
    }

    @Builder
    public Accommodation(String contentId, String title, String lodgingType, String tel, String zipcode,
                         String addr, String checkinTime, String checkoutTime, String roomCount,
                         String roomType, String parking, String subFacility, String overview,
                         String firstImage, String firstImage2, String mapX, String mapY,
                         Member host, AccommodationStatus approvalStatus, String rejectReason) {
        this.contentId = contentId;
        this.title = title;
        this.lodgingType = lodgingType;
        this.tel = tel;
        this.zipcode = zipcode;
        this.addr = addr;
        this.checkinTime = checkinTime;
        this.checkoutTime = checkoutTime;
        this.roomCount = roomCount;
        this.roomType = roomType;
        this.parking = parking;
        this.subFacility = subFacility;
        this.overview = overview;
        this.firstImage = firstImage;
        this.firstImage2 = firstImage2;
        this.mapX = mapX;
        this.mapY = mapY;
        this.host = host;
        this.approvalStatus = (approvalStatus != null) ? approvalStatus : AccommodationStatus.PENDING;
        this.rejectReason = rejectReason;
    }
}

package org.example.hotelbooking.domain.accommodation.repository;

import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.accommodation.dto.StayResponseDto;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

import java.util.List;

public interface AccommodationRepository extends JpaRepository<Accommodation, String>, AccommodationRepositoryCustom {

    @Query("SELECT new org.example.hotelbooking.domain.accommodation.dto.StayResponseDto(" +
            "a.contentId, a.host.memberId, a.title, a.lodgingType, a.tel, a.zipcode, a.addr, " +
            "a.checkinTime, a.checkoutTime, a.roomCount, a.roomType, a.parking, a.subFacility, " +
            "a.overview, a.firstImage, a.firstImage2, a.mapX, a.mapY, " +
            "COALESCE((SELECT MIN(r.pricePerNight) FROM Room r WHERE r.accommodation = a), 0)) " +
            "FROM Accommodation a " +
            "WHERE (:region IS NULL OR :region = '' OR :region = '전체' OR a.addr LIKE %:region%) " +
            "  AND (:type IS NULL OR :type = '' OR :type = '전체' OR a.lodgingType LIKE %:type%) " +
            "  AND (:query IS NULL OR :query = '' OR a.title LIKE %:query% OR a.addr LIKE %:query%) " +
            "ORDER BY a.contentId DESC")
    List<StayResponseDto> searchStays(
            @Param("region") String region,
            @Param("type") String type,
            @Param("query") String query
    );
    // 숙소 생성시 시퀀스 생성
    @Query(value = "SELECT SEQ_HOST_ACCOM.NEXTVAL FROM DUAL", nativeQuery = true)
    Long getNextHostAccomSeq();

    // 호스트가 등록한 숙소 찾아오는 jpa쿼리
    List<Accommodation> findByHost_Email(String email);

    // 숙소 관리자의 숙소 보유 여부 확인
    boolean existsByHost_MemberIdAndApprovalStatus(Long memberId, String approvalStatus);
}

package org.example.hotelbooking.domain.accommodation.repository;

import com.querydsl.jpa.impl.JPAQueryFactory;
import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.accommodation.entity.AccommodationStatus;
import org.springframework.stereotype.Repository;

import java.util.List;

// QAccommodation 임포트 (빨간 줄 뜨면 Build -> Build Project 한 번 눌러주세요)
import static org.example.hotelbooking.domain.accommodation.entity.QAccommodation.accommodation;

@Repository
@RequiredArgsConstructor
public class AccommodationRepositoryImpl implements AccommodationRepositoryCustom {

    private final JPAQueryFactory queryFactory;

    @Override
    public List<Accommodation> findAccommodationsInMap(String minX, String maxX, String minY, String maxY) {
        return queryFactory
                .selectFrom(accommodation)
                .where(
                        accommodation.approvalStatus.eq(AccommodationStatus.APPROVED),
                        accommodation.mapX.between(minX, maxX),
                        accommodation.mapY.between(minY, maxY)
                )
                .fetch();
    }
}

package org.example.hotelbooking.domain.accommodation.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.accommodation.dto.AccommodationDto;
import org.example.hotelbooking.domain.accommodation.dto.AccommodationResponse;
import org.example.hotelbooking.domain.accommodation.entity.Accommodation;
import org.example.hotelbooking.domain.accommodation.entity.AccommodationStatus;
import org.example.hotelbooking.domain.accommodation.repository.AccommodationRepository;
import org.example.hotelbooking.domain.member.entity.Member;
import org.example.hotelbooking.domain.member.repository.MemberRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
@Transactional(readOnly = true) // 전체는 읽기 전용으로 성능 최적화
public class AccommodationService {

    private final AccommodationRepository accommodationRepository;
    private final MemberRepository memberRepository; // 💡 호스트 정보를 찾기 위해 추가!

    // 기존에 만드신 지도 조회 기능 (유지)
    public List<AccommodationResponse> getAccommodationsInMap(String minX, String maxX, String minY, String maxY) {
        return accommodationRepository.findAccommodationsInMap(minX, maxX, minY, maxY)
                .stream()
                .map(AccommodationResponse::from)
                .collect(Collectors.toList());
    }

    // 💡 새롭게 추가하는 숙소 등록 기능
    @Transactional // 쓰기(Save) 작업이므로 이 메서드만 readOnly 옵션을 해제합니다.
    public String register(String email, AccommodationDto.RegisterRequest request) {

        Member host = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자 정보를 찾을 수 없습니다."));

        String contentId = request.getContentId();
        if (contentId == null || contentId.isBlank()) {
            // 💡 오라클 시퀀스에서 다음 번호(1, 2, 3...)를 가져옵니다.
            Long seqNumber = accommodationRepository.getNextHostAccomSeq();
            // 1부터 들어가기
            contentId = String.valueOf(seqNumber);
        }

        Accommodation accommodation = Accommodation.builder()
                .contentId(contentId)
                .title(request.getTitle())
                .lodgingType(request.getLodgingType())
                .tel(request.getTel())
                .zipcode(request.getZipcode())
                .addr(request.getAddr())
                .checkinTime(request.getCheckinTime())
                .checkoutTime(request.getCheckoutTime())
                .roomCount(request.getRoomCount())
                .roomType(request.getRoomType())
                .parking(request.getParking())
                .subFacility(request.getSubFacility())
                .overview(request.getOverview())
                .host(host) // JWT 토큰에서 꺼낸 유저를 숙소 주인으로 등록
                .approvalStatus(AccommodationStatus.PENDING)
                .build();

        accommodationRepository.save(accommodation);

        return accommodation.getContentId();
    }

    // 💡호스트 이메일로 내 숙소 목록 가져오기
    public List<AccommodationResponse> getMyAccommodations(String email) {
        return accommodationRepository.findByHost_Email(email)
                .stream()
                .map(AccommodationResponse::from)
                .collect(Collectors.toList());
    }

    // 숙소 관리자 숙소 보유 여부 확인(호스트 이메일로 회원을 찾아 승인된(APPROVED) 숙소가 있는지 체크)
    public boolean hasApprovedAccommodation(String email) {
        Member host = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자 정보를 찾을 수 없습니다."));

        return accommodationRepository.existsByHost_MemberIdAndApprovalStatus(host.getMemberId(), "APPROVED");

@RestController
@RequestMapping("/api/v1/auth")
@RequiredArgsConstructor

    private final AuthService authService;

    // 기존 AuthController 클래스 안에 필드 주입 추가

    public ResponseEntity<String> signup(@RequestBody AuthDto.SignupRequest request) {

    }

    public ResponseEntity<AuthDto.TokenResponse> login(@RequestBody AuthDto.LoginRequest request) {

    }

    // 카카오 로그인 간 json만 띄워서 기존 AuthController 내부의 kakaoCallback 메서드 교체
    @GetMapping("/kakao/callback")
    public ResponseEntity<Void> kakaoCallback(@RequestParam String code) {
        // 1. 카카오 로그인 후 JWT 토큰 발급
        String token = kakaoAuthService.kakaoLogin(code);

        // 2. 리액트 프론트엔드로 토큰을 쿼리 스트링에 담아 리다이렉트
        HttpHeaders headers = new HttpHeaders();
        headers.setLocation(URI.create("http://localhost:5173/oauth/callback?token=" + token));

        return new ResponseEntity<>(headers, HttpStatus.FOUND); // 302 리다이렉트
    }

    @PostMapping("/apply-host")
    public ResponseEntity<AuthDto.TokenResponse> applyHost(
            @RequestBody AuthDto.HostApplyRequest request,
            @AuthenticationPrincipal String email
    ) {
        if (email == null) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED).build();
        }

package org.example.hotelbooking.domain.member.service;

import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.member.dto.AuthDto;
import org.example.hotelbooking.domain.member.entity.Member;
import org.example.hotelbooking.domain.member.entity.Role;
import org.example.hotelbooking.domain.member.repository.MemberRepository;
import org.example.hotelbooking.domain.member.repository.RoleRepository;
import org.example.hotelbooking.global.security.JwtTokenProvider;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@RequiredArgsConstructor
public class AuthService {

    private final MemberRepository memberRepository;
    private final RoleRepository roleRepository;
    private final PasswordEncoder passwordEncoder;
    private final JwtTokenProvider jwtTokenProvider;

    @Transactional
    public void signup(AuthDto.SignupRequest request) {
        if (memberRepository.existsByEmail(request.getEmail())) {
            throw new IllegalArgumentException("이미 가입된 이메일입니다.");
        }

        // 1. 기본 권한(ROLE_USER)을 DB에서 찾아옵니다.
        // (주의: 오라클 ROLES 테이블에 'ROLE_USER' 데이터가 미리 1줄 들어있어야 합니다!)
        Role userRole = roleRepository.findByRoleName("ROLE_USER")
                .orElseThrow(() -> new IllegalArgumentException("시스템 에러: 기본 권한이 존재하지 않습니다."));

        // 2. 비밀번호를 암호화해서 Member 엔티티를 조립합니다.
        Member member = Member.builder()
                .email(request.getEmail())
                .password(passwordEncoder.encode(request.getPassword())) // 암호화!
                .name(request.getName())
                .businessNo(request.getBusinessNo())
                .role(userRole)
                .build();

        // 3. DB에 저장
        memberRepository.save(member);
    }

    @Transactional(readOnly = true)
    public AuthDto.TokenResponse login(AuthDto.LoginRequest request) {
        // 1. 이메일로 회원 찾기
        Member member = memberRepository.findByEmail(request.getEmail())
                .orElseThrow(() -> new IllegalArgumentException("가입되지 않은 이메일입니다."));

        // 2. 비밀번호 일치 여부 확인 (입력된 평문 비밀번호 vs DB의 암호화된 비밀번호)
        if (!passwordEncoder.matches(request.getPassword(), member.getPassword())) {
            throw new IllegalArgumentException("비밀번호가 일치하지 않습니다.");
        }

        // 3. 성공 시 JWT 토큰 생성 후 반환 (member.getRole().getRoleName()으로 "ROLE_USER" 추출)
        String token = jwtTokenProvider.createToken(member.getEmail(), member.getRole().getRoleName(), member.getName());

        return new AuthDto.TokenResponse(token);
    }

    // 유저에서 숙소관리자 신청 후 숙소관리자(ROLE_HOST) 권한 부여 로직
    @Transactional
    public String upgradeToHost(String email, String businessNo) {
        Member member = memberRepository.findByEmail(email)
                .orElseThrow(() -> new IllegalArgumentException("사용자를 찾을 수 없습니다."));

        Role hostRole = roleRepository.findByRoleName("ROLE_HOST")
                .orElseThrow(() -> new IllegalArgumentException("ROLE_HOST 권한이 존재하지 않습니다."));

        // 리플렉션이나 setter 없이 엔티티 업데이트 (Member 엔티티에 비즈니스 메서드가 없다면 builder 또는 편의 메서드 활용)
        // Member.java에 호스트 전환용 메서드를 추가하거나 아래처럼 필드를 변경합니다.
        member.updateToHost(businessNo, hostRole);

        return jwtTokenProvider.createToken(member.getEmail(), hostRole.getRoleName(), member.getName());
    }
}

package org.example.hotelbooking.domain.member.service;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import lombok.RequiredArgsConstructor;
import org.example.hotelbooking.domain.member.entity.Member;
import org.example.hotelbooking.domain.member.entity.Role;
import org.example.hotelbooking.domain.member.repository.MemberRepository;
import org.example.hotelbooking.domain.member.repository.RoleRepository;
import org.example.hotelbooking.global.security.JwtTokenProvider;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpEntity;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpMethod;
import org.springframework.http.ResponseEntity;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.util.LinkedMultiValueMap;
import org.springframework.util.MultiValueMap;
import org.springframework.web.client.RestTemplate;

import java.util.UUID;

@Service
@RequiredArgsConstructor
public class KakaoAuthService {

    @Value("${kakao.client.id}")
    private String clientId;

    @Value("${kakao.redirect.uri}")
    private String redirectUri;

    private final MemberRepository memberRepository;
    private final RoleRepository roleRepository;
    private final PasswordEncoder passwordEncoder;
    private final JwtTokenProvider jwtTokenProvider;

    @Transactional
    public String kakaoLogin(String code) {
        // 1. 카카오 서버로 인가 코드를 보내서 '액세스 토큰'을 받아옵니다.
        String accessToken = getAccessToken(code);

        // 2. 받아온 액세스 토큰으로 카카오 유저 정보(고유번호, 닉네임)를 가져옵니다.
        JsonNode userInfo = getUserInfo(accessToken);
        String kakaoId = userInfo.get("id").asText();
        String nickname = userInfo.get("properties").get("nickname").asText();

        // 3. 이메일을 못 받으므로, 카카오 고유번호를 이용해 가짜 이메일을 생성합니다.
        String dummyEmail = "kakao_" + kakaoId + "@hotel.com";

        // 4. DB에 이미 있는 유저인지 확인하고, 없으면 자동 회원가입 시킵니다.
        Member member = memberRepository.findByEmail(dummyEmail).orElseGet(() -> {
            Role userRole = roleRepository.findByRoleName("ROLE_USER")
                    .orElseThrow(() -> new IllegalArgumentException("기본 권한이 없습니다."));

            // 비밀번호는 랜덤 UUID로 생성해서 암호화 (어차피 카카오 로그인은 비밀번호를 안 씁니다)
            Member newMember = Member.builder()
                    .email(dummyEmail)
                    .password(passwordEncoder.encode(UUID.randomUUID().toString()))
                    .name(nickname)
                    .role(userRole)
                    .build();
            return memberRepository.save(newMember);
        });

        // 5. 회원가입(또는 로그인)이 완료되었으니, 우리 사이트 전용 JWT 토큰을 발급합니다.
        return jwtTokenProvider.createToken(member.getEmail(), member.getRole().getRoleName(), member.getName());
    }

    // --- 카카오 API 통신 메서드 ---

    private String getAccessToken(String code) {
        RestTemplate rt = new RestTemplate();
        HttpHeaders headers = new HttpHeaders();
        headers.add("Content-type", "application/x-www-form-urlencoded;charset=utf-8");

        MultiValueMap<String, String> params = new LinkedMultiValueMap<>();
        params.add("grant_type", "authorization_code");
        params.add("client_id", clientId);
        params.add("redirect_uri", redirectUri);
        params.add("code", code);

        HttpEntity<MultiValueMap<String, String>> request = new HttpEntity<>(params, headers);
        ResponseEntity<String> response = rt.postForEntity("https://kauth.kakao.com/oauth/token", request, String.class);

        try {
            ObjectMapper mapper = new ObjectMapper();
            JsonNode node = mapper.readTree(response.getBody());
            return node.get("access_token").asText();
        } catch (Exception e) {
            throw new RuntimeException("카카오 토큰 발급 중 오류가 발생했습니다.", e);
        }
    }

    private JsonNode getUserInfo(String accessToken) {
        RestTemplate rt = new RestTemplate();
        HttpHeaders headers = new HttpHeaders();
        headers.add("Authorization", "Bearer " + accessToken);
        headers.add("Content-type", "application/x-www-form-urlencoded;charset=utf-8");

        HttpEntity<MultiValueMap<String, String>> request = new HttpEntity<>(headers);
        ResponseEntity<String> response = rt.exchange("https://kapi.kakao.com/v2/user/me", HttpMethod.GET, request, String.class);

        try {
            ObjectMapper mapper = new ObjectMapper();
            return mapper.readTree(response.getBody());
        } catch (Exception e) {
            throw new RuntimeException("카카오 유저 정보 조회 중 오류가 발생했습니다.", e);
        }
    }
}

package org.example.hotelbooking.global.security;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.util.StringUtils;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.Collections;

// 💡 롬복을 써서 JwtTokenProvider를 자동으로 주입받습니다.
@RequiredArgsConstructor
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private final JwtTokenProvider jwtTokenProvider;

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
            throws ServletException, IOException {

        // 1. 사용자 요청 헤더에서 JWT 토큰을 쓱 뽑아옵니다. (아래 만들어둔 resolveToken 메서드 사용)
        String token = resolveToken(request);

        // 2. 토큰이 비어있지 않고, 진품(유효기간 안 지남 등)인지 검사합니다.
        if (token != null && jwtTokenProvider.validateToken(token)) {

            // 3. 진품이라면 토큰 안에 적혀있는 사용자 이메일(Subject)과 역할(Role)을 꺼냅니다.
            String email = jwtTokenProvider.getUserEmail(token);
            String role = jwtTokenProvider.getUserRole(token);

            // 4. 꺼낸 정보를 바탕으로 "이 사람은 인증된 사용자입니다" 라는 확인증(Authentication 객체)을 만듭니다.
            Authentication authentication = new UsernamePasswordAuthenticationToken(
                    email,
                    null,
                    Collections.singletonList(new SimpleGrantedAuthority(role))
            );

            // 5. 스프링 시큐리티의 핵심 저장소(SecurityContext)에 이 확인증을 예쁘게 보관해 둡니다.
            //    (이렇게 해두면 나중에 컨트롤러에서 @AuthenticationPrincipal 로 이 사람 정보를 꺼내 쓸 수 있습니다!)
            SecurityContextHolder.getContext().setAuthentication(authentication);
        }

        // 6. 검사가 끝났으니 다음 필터나 컨트롤러로 무사히 넘겨줍니다.
        filterChain.doFilter(request, response);
    }

    // 💡 프론트엔드가 보낸 요청(Header) 주머니에서 "Bearer xxxx.yyyy.zzzz" 형태의 토큰만 쏙 빼내는 기능
    private String resolveToken(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        // 토큰이 존재하고, "Bearer "로 시작하면 (7글자)
        if (StringUtils.hasText(bearerToken) && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7); // "Bearer " 이후의 진짜 토큰 부분만 잘라서 반환
        }
        return null; // 토큰이 없으면 null 반환
    }
}

package org.example.hotelbooking.global.security;

import io.jsonwebtoken.*;
import io.jsonwebtoken.io.Decoders;
import io.jsonwebtoken.security.Keys;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import java.security.Key;
import java.util.Date;

@Slf4j
@Component
public class JwtTokenProvider {

    private final Key key;
    private final long expirationTime;

    // yml 또는 properties 파일의 값을 불러와서 Key 객체로 만듭니다.
    public JwtTokenProvider(@Value("${jwt.secret}") String secretKey,
                            @Value("${jwt.expiration}") long expirationTime) {
        byte[] keyBytes = Decoders.BASE64.decode(secretKey);
        this.key = Keys.hmacShaKeyFor(keyBytes);
        this.expirationTime = expirationTime;
    }

    // 💡 1. 토큰 생성 (로그인 성공 시 프론트엔드로 내려줄 출입증)
    // 프론트엔드에서 로그인 시 실제 닉네임을 띄우기 위해 String name, .claim("name", name) 추가
    public String createToken(String email, String role, String name) {
        Date now = new Date();
        Date validity = new Date(now.getTime() + this.expirationTime);

        return Jwts.builder()
                .setSubject(email) // 토큰 주인을 이메일로 기록
                .claim("role", role) // 권한 정보 기록 (예: ROLE_USER)
                .claim("name", name) // 💡 토큰 claim에 DB name 추가
                .setIssuedAt(now) // 발행 시간
                .setExpiration(validity) // 만료 시간
                .signWith(key, SignatureAlgorithm.HS256) // 암호화 알고리즘
                .compact();
    }

    // 💡 2. 토큰 유효성 검증 (API 요청이 올 때마다 진짜 출입증인지 확인)
    public boolean validateToken(String token) {
        try {
            Jwts.parserBuilder().setSigningKey(key).build().parseClaimsJws(token);
            return true;
        } catch (io.jsonwebtoken.security.SecurityException | MalformedJwtException e) {
            log.info("잘못된 JWT 서명입니다.");
        } catch (ExpiredJwtException e) {
            log.info("만료된 JWT 토큰입니다.");
        } catch (UnsupportedJwtException e) {
            log.info("지원되지 않는 JWT 토큰입니다.");
        } catch (IllegalArgumentException e) {
            log.info("JWT 토큰이 잘못되었습니다.");
        }
        return false;

    }
    // 💡 3. 토큰을 분해해서 주인의 이메일(Subject)을 꺼내는 기능
    public String getUserEmail(String token) {
        return Jwts.parserBuilder().setSigningKey(key).build()
                .parseClaimsJws(token)
                .getBody()
                .getSubject();
    }

    // 💡 4. 토큰을 분해해서 주인의 권한(Role)을 꺼내는 기능
    public String getUserRole(String token) {
        return Jwts.parserBuilder().setSigningKey(key).build()
                .parseClaimsJws(token)
                .getBody()
                .get("role", String.class);
    }
}

package org.example.hotelbooking.global.security;

import lombok.RequiredArgsConstructor;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.Customizer;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;
import org.springframework.web.cors.CorsConfiguration;
import org.springframework.web.cors.CorsConfigurationSource;
import org.springframework.web.cors.UrlBasedCorsConfigurationSource;

import java.util.Arrays;
import java.util.List;

@Configuration
@EnableWebSecurity
@RequiredArgsConstructor
public class SecurityConfig {

    private final JwtTokenProvider jwtTokenProvider;

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
                // React(Vite)와 통신하기 위한 CORS 설정
                .cors(Customizer.withDefaults())

                // REST API이므로 CSRF 보호 비활성화
                .csrf(csrf -> csrf.disable())

                // JWT를 사용하므로 서버 세션을 사용하지 않음
                .sessionManagement(session ->
                        session.sessionCreationPolicy(SessionCreationPolicy.STATELESS)
                )

                // API 접근 권한 설정
                .authorizeHttpRequests(auth -> auth

                        // 로그인 / 회원가입
                        .requestMatchers("/api/v1/auth/**").permitAll()

                        // 숙소 조회
                        .requestMatchers("/api/v1/accommodations/**").permitAll()

                        // 기존 psj의 숙소 조회 API
                        .requestMatchers("/api/v1/stays/**").permitAll()

                        // 그 외 요청은 JWT 인증 필요
                        .anyRequest().authenticated()
                )

                // JWT 인증 필터 등록
                .addFilterBefore(
                        new JwtAuthenticationFilter(jwtTokenProvider),
                        UsernamePasswordAuthenticationFilter.class
                );

        return http.build();
    }

    // 비밀번호 암호화
    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    // React 개발 서버 CORS 설정
    @Bean
    public CorsConfigurationSource corsConfigurationSource() {
        CorsConfiguration configuration = new CorsConfiguration();

        configuration.setAllowedOrigins(
                List.of("http://localhost:5173")
        );

        configuration.setAllowedMethods(
                Arrays.asList(
                        "GET",
                        "POST",
                        "PUT",
                        "DELETE",
                        "OPTIONS",
                        "PATCH"
                )
        );

        configuration.setAllowedHeaders(List.of("*"));
        configuration.setAllowCredentials(true);
        configuration.setMaxAge(3600L);

        UrlBasedCorsConfigurationSource source =
                new UrlBasedCorsConfigurationSource();

        source.registerCorsConfiguration("/**", configuration);

        return source;
    }
}

import type { Booking, BookingResponse, Stay } from '../types/stay';

const apiBase = (import.meta.env.VITE_API_BASE_URL || '').replace(/\/$/, '');

export async function fetchStays(): Promise<Stay[]> {
  const response = await fetch(`${apiBase}/api/v1/stays`);
  if (!response.ok) throw new Error('숙소 정보를 불러오지 못했습니다.');
  const stays: unknown = await response.json();
  if (!Array.isArray(stays)) throw new Error('숙소 응답 형식이 올바르지 않습니다.');
  return stays as Stay[];
}

export async function createReservation(booking: Booking): Promise<BookingResponse> {
  const response = await fetch(`${apiBase}/api/v1/reservations`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(booking),
  });
  const data: BookingResponse = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(data.message || '예약 요청을 처리하지 못했습니다.');
  return data;
}

export function useHeroSlider() {
  const [slide, setSlide] = useState(0);
  useEffect(() => {
    const timer = window.setInterval(() => setSlide(current => (current + 1) % slideCount), 10000);
    return () => window.clearInterval(timer);
  }, []);
  const previous = () => setSlide(current => (current + slideCount - 1) % slideCount);
  const next = () => setSlide(current => (current + 1) % slideCount);
  return { slide, previous, next };

import { useEffect, useState, type FormEvent } from 'react';
import { createReservation } from '../api/stays';
import type { Booking, Stay } from '../types/stay';

export function useReservation(apiOnline: boolean) {
  const [selected, setSelected] = useState<Stay | null>(null);
  const [checkIn, setCheckIn] = useState('');
  const [checkOut, setCheckOut] = useState('');
  const [guests, setGuests] = useState(2);
  const [guestName, setGuestName] = useState('');
  const [phone, setPhone] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [feedback, setFeedback] = useState('');

  useEffect(() => {
    document.body.style.overflow = selected ? 'hidden' : '';
    return () => { document.body.style.overflow = ''; };
  }, [selected]);

  const open = (stay: Stay) => { setSelected(stay); setFeedback(''); };
  const close = () => { setSelected(null); setFeedback(''); };

  const book = async (event: FormEvent) => {
    event.preventDefault();
    if (!selected) return;
    if (checkOut <= checkIn) {
      setFeedback('퇴실일은 입실일 다음 날부터 선택해 주세요.');
      return;
    }
    if (!apiOnline) {
      setFeedback('현재 예약 접수를 준비 중입니다.');
      return;
    }
    setSubmitting(true);
    setFeedback('');
    try {
      const booking: Booking = {
        contentId: selected.contentId, // 수정: contentId 바인딩
        checkIn,
        checkOut,
        guests,
        guestName,
        phone
      };
      const data = await createReservation(booking);
      setFeedback(`예약 요청이 접수되었습니다. 예약번호: ${data.reservationCode || data.id}`);
    } catch (error) {
      setFeedback(error instanceof Error ? error.message : '잠시 후 다시 시도해 주세요.');
    } finally {
      setSubmitting(false);
    }
  };

  return { selected, open, close, checkIn, setCheckIn, checkOut, setCheckOut, guests, setGuests,
    guestName, setGuestName, phone, setPhone, submitting, feedback, book };
}

import { useEffect, useState } from 'react';
import { fetchStays } from '../data/stays';
import type { Stay } from '../types/stay';

export function useStays() {
  const [region, setRegion] = useState('전체 지역');
  const [type, setType] = useState('전체 숙박');
  const [query, setQuery] = useState('');
  const [term, setTerm] = useState('');
  const [stays, setStays] = useState<Stay[]>([]);
  const [apiOnline, setApiOnline] = useState(false);
  const [loading, setLoading] = useState(true);

  // 선택된 지역, 유형, 검색어가 바뀔 때마다 스프링 API 호출
  useEffect(() => {
    let active = true;
    setLoading(true);

    fetchStays(region, type, term)
        .then(data => {
          if (active) {
            setStays(data);
            setApiOnline(true);
          }
        })
        .catch((err) => {
          console.error('숙소 로딩 실패:', err);
          if (active) setApiOnline(false);
        })
        .finally(() => {
          if (active) setLoading(false);
        });

    return () => { active = false; };
  }, [region, type, term]);

  const selectRegion = (value: string) => {
    setRegion(value);
    document.getElementById('stays')?.scrollIntoView({ behavior: 'smooth', block: 'start' });
  };

  const search = () => {
    setTerm(query);
    document.getElementById('stays')?.scrollIntoView({ behavior: 'smooth' });
  };

  return {
    region,
    setRegion,
    type,
    setType,
    query,
    setQuery,
    result: stays,
    loading,
    apiOnline,
    selectRegion,
    search,
  };
}

import { useEffect } from 'react';
import { useNavigate, useSearchParams } from 'react-router-dom';

export function KakaoCallback() {
    const [searchParams] = useSearchParams();
    const navigate = useNavigate();

    useEffect(() => {
        const token = searchParams.get('token');
        if (token) {
            // 발급된 JWT 토큰을 로컬스토리지에 저장
            localStorage.setItem('accessToken', token);
            alert('카카오 로그인에 성공했습니다!');
            navigate('/'); // 메인 화면으로 이동
        } else {
            alert('로그인에 실패했습니다.');
            navigate('/login');
        }
    }, [searchParams, navigate]);

    return (
        <div style={{ textAlign: 'center', marginTop: '100px', fontSize: '18px' }}>
            로그인 처리 중입니다. 잠시만 기다려주세요...
        </div>
    );
